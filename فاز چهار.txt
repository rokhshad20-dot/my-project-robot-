def verify_password(plain_password: str, hashed_password: str) -> bool:
    """بررسی رمز عبور"""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """Hash کردن رمز عبور"""
    return pwd_context.hash(password)

def get_user(username: str) -> Optional[UserInDB]:
    """دریافت کاربر از دیتابیس"""
    if username in USERS_DB:
        user_dict = USERS_DB[username]
        return UserInDB(**user_dict)
    return None

def authenticate_user(username: str, password: str) -> Optional[UserInDB]:
    """احراز هویت کاربر"""
    # بررسی قفل
    if username in failed_attempts:
        lockout_until, attempts = failed_attempts[username]
        if datetime.now() < lockout_until:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"حساب قفل شده. {(lockout_until - datetime.now()).seconds // 60} دقیقه صبر کنید"
            )
    
    user = get_user(username)
    if not user:
        return None
    
    if not verify_password(password, user.hashed_password):
        # ثبت تلاش ناموفق
        if username not in failed_attempts:
            failed_attempts[username] = [datetime.now(), 1]
        else:
            lockout_until, attempts = failed_attempts[username]
            attempts += 1
            failed_attempts[username] = [lockout_until, attempts]
            
            # قفل بعد از 5 تلاش
            if attempts >= 5:
                lockout_until = datetime.now() + timedelta(minutes=LOCKOUT_DURATION)
                failed_attempts[username] = [lockout_until, attempts]
        
        return None
    
    # پاک کردن تلاش‌های ناموفق
    if username in failed_attempts:
        del failed_attempts[username]
    
    return user

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    """ساخت Access Token"""
    to_encode = data.copy()
    
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def create_refresh_token(data: dict) -> str:
    """ساخت Refresh Token"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

async def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    """دریافت کاربر فعلی از Token"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="اطلاعات احراز هویت نامعتبر است",
        headers={"WWW-Authenticate": "Bearer"},
    )
    
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise credentials_exception
        token_data = TokenData(username=username)
    except jwt.PyJWTError:
        raise credentials_exception
    
    user = get_user(username=token_data.username)
    if user is None:
        raise credentials_exception
    
    return user

@app.post("/api/token", response_model=Token)
async def login(form_data: OAuth2PasswordRequestForm = Depends()):
    """ورود و دریافت Token"""
    user = authenticate_user(form_data.username, form_data.password)
    
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="نام کاربری یا رمز عبور اشتباه است",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    # ساخت Tokens
    access_token = create_access_token(data={"sub": user.username})
    refresh_token = create_refresh_token(data={"sub": user.username})
    
    return {
        "access_token": access_token,
        "refresh_token": refresh_token,
        "token_type": "bearer"
    }

@app.post("/api/token/refresh", response_model=Token)
async def refresh_token(refresh_token: str):
    """تمدید Token با Refresh Token"""
    try:
        payload = jwt.decode(refresh_token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username is None:
            raise HTTPException(status_code=401, detail="Token نامعتبر")
        
        # ساخت Token جدید
        access_token = create_access_token(data={"sub": username})
        new_refresh_token = create_refresh_token(data={"sub": username})
        
        return {
            "access_token": access_token,
            "refresh_token": new_refresh_token,
            "token_type": "bearer"
        }
    
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail="Token منقضی شده")

@app.post("/api/users/change-password")
async def change_password(
    password_data: PasswordChange,
    current_user: User = Depends(get_current_user)
):
    """تغییر رمز عبور"""
    user_db = get_user(current_user.username)
    
    # بررسی رمز فعلی
    if not verify_password(password_data.current_password, user_db.hashed_password):
        raise HTTPException(status_code=400, detail="رمز عبور فعلی اشتباه است")
    
    # تغییر رمز
    USERS_DB[current_user.username]["hashed_password"] = get_password_hash(password_data.new_password)
    
    return {"message": "رمز عبور با موفقیت تغییر کرد"}

class BotManager:
    """
    مدیر ربات - کنترل Start/Stop/Status
    
    استفاده:
        manager = BotManager()
        manager.start()
        manager.stop()
        status = manager.get_status()
    """
    
    def __init__(self):
        self.status_file = Path("data/bot_status.json")
        self.pid_file = Path("data/bot.pid")
        self.config_file = Path("config.py")
        
        # ساخت پوشه
        self.status_file.parent.mkdir(parents=True, exist_ok=True)
        
        # بارگذاری وضعیت
        self._load_status()
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # کنترل ربات
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    def start(self) -> Dict:
        """
        شروع ربات
        
        Returns:
            Dict با وضعیت
        """
        # بررسی اگر قبلاً در حال اجراست
        if self.is_running():
            return {
                "success": False,
                "message": "ربات در حال اجراست",
                "pid": self._get_pid()
            }
        
        try:
            # اجرای ربات به عنوان subprocess
            process = subprocess.Popen(
                ['python', 'bot_with_dynamic_sl.py'],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                creationflags=subprocess.CREATE_NEW_PROCESS_GROUP if hasattr(subprocess, 'CREATE_NEW_PROCESS_GROUP') else 0
            )
            
            # ذخیره PID
            self._save_pid(process.pid)
            
            # ذخیره وضعیت
            self._save_status({
                "running": True,
                "pid": process.pid,
                "start_time": datetime.now().isoformat(),
                "mode": "live"
            })
            
            return {
                "success": True,
                "message": "ربات شروع شد",
                "pid": process.pid
            }
        
        except Exception as e:
            return {
                "success": False,
                "message": f"خطا در شروع: {str(e)}"
            }
    
    def stop(self) -> Dict:
        """
        توقف ربات
        
        Returns:
            Dict با وضعیت
        """
        if not self.is_running():
            return {
                "success": False,
                "message": "ربات در حال اجرا نیست"
            }
        
        try:
            pid = self._get_pid()
            
            # Kill کردن Process
            if pid:
                try:
                    process = psutil.Process(pid)
                    process.terminate()  # SIGTERM (Graceful)
                    
                    # صبر 5 ثانیه
                    time.sleep(5)
                    
                    # اگر هنوز زنده است، Kill سخت
                    if process.is_running():
                        process.kill()  # SIGKILL (Force)
                
                except psutil.NoSuchProcess:
                    pass  # قبلاً بسته شده
            
            # پاک کردن فایل‌ها
            if self.pid_file.exists():
                self.pid_file.unlink()
            
            # ذخیره وضعیت
            self._save_status({
                "running": False,
                "pid": None,
                "stop_time": datetime.now().isoformat()
            })
            
            return {
                "success": True,
                "message": "ربات متوقف شد"
            }
        
        except Exception as e:
            return {
                "success": False,
                "message": f"خطا در توقف: {str(e)}"
            }
    
    def restart(self) -> Dict:
        """
        Restart ربات
        
        Returns:
            Dict با وضعیت
        """
        # ابتدا Stop
        stop_result = self.stop()
        
        if not stop_result["success"]:
            return stop_result
        
        # صبر 2 ثانیه
        time.sleep(2)
        
        # حالا Start
        return self.start()
    
    def is_running(self) -> bool:
        """
        بررسی اجرای ربات
        
        Returns:
            True اگر در حال اجراست
        """
        pid = self._get_pid()
        
        if not pid:
            return False
        
        try:
            process = psutil.Process(pid)
            return process.is_running()
        except psutil.NoSuchProcess:
            return False
    
    def get_status(self) -> Dict:
        """
        دریافت وضعیت کامل ربات
        
        Returns:
            Dict شامل وضعیت، uptime، و ...
        """
        status = self._load_status()
        
        # بررسی واقعی
        actually_running = self.is_running()
        
        # اگر فرق داشت، اصلاح کن
        if status.get("running") != actually_running:
            status["running"] = actually_running
            self._save_status(status)
        
        # محاسبه uptime
        if actually_running and "start_time" in status:
            start_time = datetime.fromisoformat(status["start_time"])
            uptime_seconds = (datetime.now() - start_time).total_seconds()
            status["uptime_seconds"] = int(uptime_seconds)
            status["uptime_formatted"] = self._format_uptime(uptime_seconds)
        else:
            status["uptime_seconds"] = 0
            status["uptime_formatted"] = "0h 0m"
        
        return status

        except Exception as e:

            }
    
    def get_config(self) -> Dict:

        except Exception as e:

    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    def _save_pid(self, pid: int):
        """ذخیره PID"""
        with open(self.pid_file, 'w') as f:
            f.write(str(pid))
    
    def _get_pid(self) -> Optional[int]:
        """دریافت PID"""
        if not self.pid_file.exists():
            return None
        
        try:
            with open(self.pid_file, 'r') as f:
                return int(f.read().strip())
        except:
            return None
    
    def _save_status(self, status: Dict):
        """ذخیره وضعیت"""
        with open(self.status_file, 'w') as f:
            json.dump(status, f, indent=2)
    
    def _load_status(self) -> Dict:
        """بارگذاری وضعیت"""
        if not self.status_file.exists():
            return {
                "running": False,
                "pid": None
            }
        
        try:
            with open(self.status_file, 'r') as f:
                return json.load(f)
        except:
            return {
                "running": False,
                "pid": None
            }
    
    def _format_uptime(self, seconds: float) -> str:
        """فرمت uptime"""
        hours = int(seconds // 3600)
        minutes = int((seconds % 3600) // 60)
        return f"{hours}h {minutes}m"

                async init() {
                    // Check auth
                    const token = localStorage.getItem('access_token');
                    if (!token) {
                        window.location.href = '/login';
                        return;
                    }
                    
                    // Load data
                    await this.loadUser();
                    await this.refreshData();
                    
                    // Setup chart
                    this.setupChart();
                    
                    // Auto-refresh every 5 seconds
                    setInterval(() => this.refreshData(), 5000);
                },
                
                // Load user info
                async loadUser() {
                    try {
                        const response = await this.apiCall('/api/users/me');
                        this.user = response;
                    } catch (error) {
                        console.error('Failed to load user:', error);
                        this.logout();
                    }
                },
                
                // Refresh all data
                async refreshData() {
                    this.loading = true;
                    try {
                        await Promise.all([
                            this.loadStatus(),
                            this.loadBotStatus(),
                            this.loadConfig(),
                            this.loadLogs()
                        ]);
                    } finally {
                        this.loading = false;
                    }
                },
                
                // Load bot status from bot_manager
                async loadBotStatus() {
                    try {
                        const data = await this.apiCall('/api/bot/status');
                        this.botStatus = data;
                    } catch (error) {
                        console.error('Failed to load bot status:', error);
                    }
                },
                
                // Load bot status
                async loadStatus() {
                    const data = await this.apiCall('/api/status');
                    this.status = data;
                },
                
                // Load config
                async loadConfig() {
                    const data = await this.apiCall('/api/config');
                    this.config = data;
                },
                
                // Save config
                async saveConfig() {
                    try {
                        this.loading = true;
                        const data = await this.apiCall('/api/config', 'PUT', this.config);
                        
                        if (data.needs_restart) {
                            if (confirm('⚠️ ربات در حال اجراست. Restart کنیم؟')) {
                                await this.restartBot();
                            }
                        }
                        
                        alert('✅ تنظیمات ذخیره شد');
                    } catch (error) {
                        alert('❌ خطا در ذخیره');
                    } finally {
                        this.loading = false;
                    }
                },
                
                // Bot control functions
                async startBot() {
                    if (!confirm('آیا مطمئنید که می‌خواهید ربات را شروع کنید؟')) return;
                    
                    this.loading = true;
                    try {
                        const result = await this.apiCall('/api/bot/start', 'POST');
                        if (result.success) {
                            alert('✅ ' + result.message);
                            await this.loadBotStatus();
                        } else {
                            alert('❌ ' + result.message);
                        }
                    } catch (error) {
                        alert('❌ خطا: ' + error.message);
                    } finally {
                        this.loading = false;
                    }
                },
                
                async stopBot() {
                    if (!confirm('⚠️ آیا مطمئنید که می‌خواهید ربات را متوقف کنید؟')) return;
                    
                    this.loading = true;
                    try {
                        const result = await this.apiCall('/api/bot/stop', 'POST');
                        if (result.success) {
                            alert('✅ ' + result.message);
                            await this.loadBotStatus();
                        } else {
                            alert('❌ ' + result.message);
                        }
                    } catch (error) {
                        alert('❌ خطا: ' + error.message);
                    } finally {
                        this.loading = false;
                    }
                },
                
                async restartBot() {
                    if (!confirm('🔄 آیا مطمئنید که می‌خواهید ربات را Restart کنید؟')) return;
                    
                    this.loading = true;
                    try {
                        const result = await this.apiCall('/api/bot/restart', 'POST');
                        if (result.success) {
                            alert('✅ ' + result.message);
                            await this.loadBotStatus();
                        } else {
                            alert('❌ ' + result.message);
                        }

                        alert('❌ خطا: ' + error.message);
                    } finally {
                        this.loading = false;
                    }

                    this.recentLogs = data.logs || [];
                },
                
                // API Call with JWT
                async apiCall(url, method = 'GET', body = null) {
                    const token = localStorage.getItem('access_token');
                    
                    const options = {
                        method,
                        headers: {
                            'Authorization': `Bearer ${token}`,
                            'Content-Type': 'application/json'
                        }
                    };
                    
                    if (body) {
                        options.body = JSON.stringify(body);
                    }
                    
                    const response = await fetch(url, options);
                    
                    if (response.status === 401) {
                        // Token expired
                        this.logout();
                        throw new Error('Unauthorized');
                    }
                    
                    return response.json();
                },

        function loginApp() {
            return {
                username: '',
                password: '',
                rememberMe: false,
                showPassword: false,
                loading: false,
                error: '',
                lockout: false,
                lockoutMinutes: 0,
                
                async login() {
                    this.loading = true;
                    this.error = '';
                    this.lockout = false;
                    
                    try {
                        // ساخت FormData
                        const formData = new FormData();
                        formData.append('username', this.username);
                        formData.append('password', this.password);
                        
                        // ارسال درخواست
                        const response = await fetch('/api/token', {
                            method: 'POST',
                            body: formData
                        });
                        
                        const data = await response.json();
                        
                        if (response.ok) {
                            // ذخیره Tokens
                            localStorage.setItem('access_token', data.access_token);
                            localStorage.setItem('refresh_token', data.refresh_token);
                            
                            if (this.rememberMe) {
                                localStorage.setItem('remember_me', 'true');
                            }
                            
                            // انیمیشن موفقیت
                            this.showSuccessAnimation();
                            
                            // هدایت به داشبورد
                            setTimeout(() => {
                                window.location.href = '/';
                            }, 1000);
                            
                        } else {
                            // خطا
                            if (response.status === 403) {
                                // قفل شده
                                this.lockout = true;
                                const lockoutMatch = data.detail.match(/(\d+) دقیقه/);
                                this.lockoutMinutes = lockoutMatch ? lockoutMatch[1] : 30;
                            } else {
                                this.error = data.detail || 'خطا در ورود';
                            }
                        }
                        
                    } catch (error) {
                        console.error('Login error:', error);
                        this.error = 'خطا در اتصال به سرور';
                    } finally {
                        this.loading = false;
                    }
                },
                
                showSuccessAnimation() {
                    // می‌توانید اینجا انیمیشن موفقیت اضافه کنید
                    console.log('✅ ورود موفق!');
                }
            }
        }
