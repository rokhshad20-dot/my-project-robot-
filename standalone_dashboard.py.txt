@dataclass
class AppConfig:
    """تنظیمات اپلیکیشن"""
    
    # Trading
    symbol: str = "EURUSD"
    timeframe: str = "H1"
    risk_per_trade: float = 0.01
    max_positions: int = 2
    
    # MT5
    mt5_path: Optional[str] = None
    mt5_login: Optional[int] = None
    mt5_password: Optional[str] = None
    mt5_server: Optional[str] = None
    
    # Model
    model_path: str = "data/models/ensemble"
    use_ensemble: bool = True
    
    # System
    check_interval: int = 30  # seconds
    log_level: str = "INFO"
    
    # Dashboard
    dashboard_port: int = 8000
    dashboard_host: str = "0.0.0.0"
    
    def save(self, filepath: str = "config.json"):
        """ذخیره تنظیمات"""
        with open(filepath, 'w') as f:
            json.dump(asdict(self), f, indent=2)
        print(f"✅ تنظیمات ذخیره شد: {filepath}")
    
    @classmethod
    def load(cls, filepath: str = "config.json") -> 'AppConfig':
        """بارگذاری تنظیمات"""
        if not os.path.exists(filepath):
            print("⚠️ فایل تنظیمات یافت نشد - استفاده از پیش‌فرض")
            return cls()
        
        with open(filepath, 'r') as f:
            data = json.load(f)
        print(f"✅ تنظیمات بارگذاری شد: {filepath}")
        return cls(**data)

# 🤖 BOT STATE MANAGEMENT
# ═══════════════════════════════════════════════════════════════════════════════

class BotState(Enum):
    """وضعیت ربات"""
    STOPPED = "stopped"
    STARTING = "starting"
    RUNNING = "running"
    PAUSED = "paused"
    STOPPING = "stopping"
    ERROR = "error"


@dataclass
class BotStatus:
    """وضعیت فعلی ربات"""
    state: str
    uptime_seconds: float = 0
    total_signals: int = 0
    total_trades: int = 0
    open_positions: int = 0
    equity: float = 0.0
    balance: float = 0.0
    profit: float = 0.0
    last_check: Optional[str] = None
    last_error: Optional[str] = None
    
    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

# 📊 TRADING BOT CORE
# ═══════════════════════════════════════════════════════════════════════════════

class TradingBotCore:
    """هسته اصلی ربات معامله‌گر"""
    
    def __init__(self, config: AppConfig):
        self.config = config
        self.state = BotState.STOPPED
        self.start_time: Optional[datetime] = None
        
        # Stats
        self.stats = {
            'total_signals': 0,
            'total_trades': 0,
            'winning_trades': 0,
            'losing_trades': 0
        }
        
        # MT5
        self.mt5_initialized = False
        
        # Model (lazy loading)
        self.model = None
        self.scaler = None
        
        # Logging
        self.setup_logging()
        
        # Thread
        self.thread: Optional[threading.Thread] = None
        self.stop_event = threading.Event()
    
    def setup_logging(self):
        """راه‌اندازی logging"""
        log_dir = Path("data/logs")
        log_dir.mkdir(parents=True, exist_ok=True)
        
        logging.basicConfig(
            level=getattr(logging, self.config.log_level),
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
            handlers=[
                logging.FileHandler(log_dir / f"bot_{datetime.now().strftime('%Y%m%d')}.log"),
                logging.StreamHandler()
            ]
        )
        self.logger = logging.getLogger("TradingBot")
    
    def initialize_mt5(self) -> bool:
        """راه‌اندازی MT5"""
        if self.mt5_initialized:
            return True
        
        try:
            import MetaTrader5 as mt5
            
            self.logger.info("🔌 اتصال به MetaTrader 5...")
            
            # راه‌اندازی
            if self.config.mt5_path:
                success = mt5.initialize(
                    path=self.config.mt5_path,
                    login=self.config.mt5_login,
                    password=self.config.mt5_password,
                    server=self.config.mt5_server
                )
            else:
                success = mt5.initialize()
            
            if not success:
                error = mt5.last_error()
                self.logger.error(f"❌ خطا در اتصال MT5: {error}")
                return False
            
            self.mt5_initialized = True
            self.logger.info("✅ MetaTrader 5 متصل شد")
            
            # نمایش اطلاعات حساب
            account = mt5.account_info()
            if account:
                self.logger.info(f"📊 حساب: {account.login}")
                self.logger.info(f"💰 موجودی: ${account.balance:,.2f}")
            
            return True
            
        except Exception as e:
            self.logger.error(f"❌ خطا: {e}")
            return False
    
    def shutdown_mt5(self):
        """قطع اتصال MT5"""
        if not self.mt5_initialized:
            return
        
        try:
            import MetaTrader5 as mt5
            mt5.shutdown()
            self.mt5_initialized = False
            self.logger.info("🔌 اتصال MT5 قطع شد")
        except Exception as e:
            self.logger.error(f"خطا در قطع اتصال: {e}")
    
    def load_model(self) -> bool:
        """بارگذاری مدل ML"""
        try:
            self.logger.info(f"📥 بارگذاری مدل از {self.config.model_path}...")
            
            # بارگذاری scaler
            scaler_path = Path(self.config.model_path) / "scaler_tabular.pkl"
            if scaler_path.exists():
                self.scaler = joblib.load(scaler_path)
                self.logger.info("✅ Scaler بارگذاری شد")
            
            # بارگذاری مدل (ساده‌شده برای مثال)
            # در نسخه واقعی: load ensemble model
            self.model = "MOCK_MODEL"  # جایگزین با بارگذاری واقعی
            
            self.logger.info("✅ مدل بارگذاری شد")
            return True
            
        except Exception as e:
            self.logger.error(f"❌ خطا در بارگذاری مدل: {e}")
            return False
    
    def start(self) -> bool:
        """شروع ربات"""
        if self.state != BotState.STOPPED:
            self.logger.warning("⚠️ ربات قبلاً شروع شده است")
            return False
        
        self.state = BotState.STARTING
        self.logger.info("🚀 شروع ربات...")
        
        # راه‌اندازی MT5
        if not self.initialize_mt5():
            self.state = BotState.ERROR
            return False
        
        # بارگذاری مدل
        if not self.load_model():
            self.state = BotState.ERROR
            return False
        
        # شروع thread
        self.stop_event.clear()
        self.thread = threading.Thread(target=self._run_loop, daemon=True)
        self.thread.start()
        
        self.start_time = datetime.now()
        self.state = BotState.RUNNING
        self.logger.info("✅ ربات آماده است")
        
        return True
    
    def stop(self):
        """توقف ربات"""
        self.logger.info("⏹️ توقف ربات...")
        self.state = BotState.STOPPING
        
        # توقف thread
        self.stop_event.set()
        if self.thread:
            self.thread.join(timeout=5)
        
        # قطع MT5
        self.shutdown_mt5()
        
        self.state = BotState.STOPPED
        self.start_time = None
        self.logger.info("✅ ربات متوقف شد")
    
    def pause(self):
        """مکث ربات"""
        if self.state == BotState.RUNNING:
            self.state = BotState.PAUSED
            self.logger.info("⏸️ ربات متوقف موقت شد")
    
    def resume(self):
        """ادامه ربات"""
        if self.state == BotState.PAUSED:
            self.state = BotState.RUNNING
            self.logger.info("▶️ ربات ادامه یافت")
    
    def _run_loop(self):
        """حلقه اصلی ربات"""
        self.logger.info("🔄 حلقه اصلی شروع شد")
        
        while not self.stop_event.is_set():
            try:
                if self.state == BotState.RUNNING:
                    self._run_once()
                
                # صبر تا چک بعدی
                self.stop_event.wait(self.config.check_interval)
                
            except Exception as e:
                self.logger.error(f"❌ خطا در حلقه اصلی: {e}")
                self.state = BotState.ERROR
                time.sleep(60)  # صبر قبل از تلاش دوباره
        
        self.logger.info("🔄 حلقه اصلی متوقف شد")
    
    def _run_once(self):
        """یک چرخه معاملاتی"""
        try:
            import MetaTrader5 as mt5
            
            # دریافت tick
            tick = mt5.symbol_info_tick(self.config.symbol)
            if not tick:
                return
            
            self.logger.debug(f"📊 {self.config.symbol}: Bid={tick.bid:.5f}, Ask={tick.ask:.5f}")
            
            # دریافت داده
            # در نسخه واقعی: get historical data, calculate indicators, predict
            
            # فقط برای نمایش:
            self.stats['total_signals'] += 1
            
            # شبیه‌سازی: تصادفی معامله
            if np.random.random() < 0.05:  # 5% احتمال معامله
                self.stats['total_trades'] += 1
                self.logger.info(f"💹 معامله #{self.stats['total_trades']} باز شد")
            
        except Exception as e:
            self.logger.error(f"خطا در چرخه: {e}")
    
    def get_status(self) -> BotStatus:
        """دریافت وضعیت فعلی"""
        try:
            import MetaTrader5 as mt5
            
            # محاسبه uptime
            uptime = 0.0
            if self.start_time:
                uptime = (datetime.now() - self.start_time).total_seconds()
            
            # اطلاعات حساب
            equity = 0.0
            balance = 0.0
            if self.mt5_initialized:
                account = mt5.account_info()
                if account:
                    equity = account.equity
                    balance = account.balance
            
            # پوزیشن‌های باز
            open_positions = 0
            profit = 0.0
            if self.mt5_initialized:
                positions = mt5.positions_get(symbol=self.config.symbol)
                if positions:
                    open_positions = len(positions)
                    profit = sum(p.profit for p in positions)
            
            return BotStatus(
                state=self.state.value,
                uptime_seconds=uptime,
                total_signals=self.stats['total_signals'],
                total_trades=self.stats['total_trades'],
                open_positions=open_positions,
                equity=equity,
                balance=balance,
                profit=profit,
                last_check=datetime.now().isoformat()
            )
            
        except Exception as e:
            self.logger.error(f"خطا در get_status: {e}")
            return BotStatus(
                state=self.state.value,
                last_error=str(e)
            )

@app.post("/api/start")
async def start_bot():
    """شروع ربات"""
    if not bot:
        raise HTTPException(status_code=503, detail="Bot not initialized")
    
    if bot.start():
        # اطلاع به WebSocket clients
        await broadcast_message({"type": "bot_started"})
        return {"status": "success", "message": "Bot started"}
    else:
        raise HTTPException(status_code=500, detail="Failed to start bot")

@app.put("/api/config")
async def update_config(updates: Dict[str, Any]):
    """به‌روزرسانی تنظیمات"""
    if not config or not bot:
        raise HTTPException(status_code=503, detail="Not initialized")
    
    # به‌روزرسانی
    for key, value in updates.items():
        if hasattr(config, key):
            setattr(config, key, value)
    
    # ذخیره
    config.save()
    
    # اطلاع به clients
    await broadcast_message({"type": "config_updated", "config": asdict(config)})
    
    return {"status": "success", "message": "Config updated"}


@app.get("/api/logs/recent")
async def get_recent_logs(lines: int = 100):
    """دریافت لاگ‌های اخیر"""
    log_file = Path("data/logs") / f"bot_{datetime.now().strftime('%Y%m%d')}.log"
    
    if not log_file.exists():
        return {"logs": []}
    
    try:
        with open(log_file, 'r') as f:
            all_lines = f.readlines()
            recent_lines = all_lines[-lines:]
            return {"logs": [line.strip() for line in recent_lines]}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    """WebSocket برای real-time updates"""
    await websocket.accept()
    websocket_clients.append(websocket)
    
    try:
        while True:
            # ارسال status هر 2 ثانیه
            if bot:
                status = bot.get_status()
                await websocket.send_json(status.to_dict())
            
            await asyncio.sleep(2)
            
    except WebSocketDisconnect:
        websocket_clients.remove(websocket)


async def broadcast_message(message: Dict[str, Any]):
    """ارسال پیام به تمام WebSocket clients"""
    for client in websocket_clients:
        try:
            await client.send_json(message)
        except:
            pass

                updateLogs('✅ Connected to bot');
            };
            
            ws.onmessage = (event) => {
                statusData = JSON.parse(event.data);
                updateUI(statusData);

            ws.onerror = (error) => {
                console.error('WebSocket error:', error);
                updateLogs('❌ Connection error');
            };

                updateLogs('🔌 Disconnected - Reconnecting...');

                'stopped': '⚪',
                'running': '🟢',
                'paused': '🟡',
