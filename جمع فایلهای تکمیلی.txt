/**
 * ====================================================================
 * فایل: frontend/assets/js/theme.js
 * سیستم Dark Mode با ذخیره‌سازی خودکار
 * ====================================================================
 */

class ThemeManager {
    constructor() {
        this.currentTheme = this.getStoredTheme() || 'light';
        this.init();
    }
    
    /**
     * راه‌اندازی اولیه
     */
    init() {
        // اعمال تم ذخیره شده
        this.applyTheme(this.currentTheme);
        
        // رویداد تغییر تم سیستم
        window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
            if (!this.getStoredTheme()) {
                this.applyTheme(e.matches ? 'dark' : 'light');
            }
        });
    }
    
    /**
     * دریافت تم ذخیره شده
     */
    getStoredTheme() {
        return localStorage.getItem('theme');
    }
    
    /**
     * ذخیره تم
     */
    setStoredTheme(theme) {
        localStorage.setItem('theme', theme);
    }
    
    /**
     * اعمال تم
     */
    applyTheme(theme) {
        this.currentTheme = theme;
        
        if (theme === 'dark') {
            document.documentElement.classList.add('dark');
        } else {
            document.documentElement.classList.remove('dark');
        }
        
        this.setStoredTheme(theme);
        this.updateIcon();
    }
    
    /**
     * تغییر تم (Toggle)
     */
    toggle() {
        const newTheme = this.currentTheme === 'light' ? 'dark' : 'light';
        this.applyTheme(newTheme);
    }
    
    /**
     * به‌روزرسانی آیکون
     */
    updateIcon() {
        const icon = document.getElementById('theme-icon');
        if (icon) {
            if (this.currentTheme === 'dark') {
                icon.classList.remove('fa-moon');
                icon.classList.add('fa-sun');
            } else {
                icon.classList.remove('fa-sun');
                icon.classList.add('fa-moon');
            }
        }
    }
    
    /**
     * دریافت تم فعلی
     */
    getTheme() {
        return this.currentTheme;
    }
}

// ساخت instance گلوبال
const themeManager = new ThemeManager();

// Export برای استفاده در سایر فایل‌ها
if (typeof module !== 'undefined' && module.exports) {
    module.exports = ThemeManager;
}

    <script>
        function dashboardApp() {
            return {
                // State
                sidebarOpen: true,
                currentPage: 'overview',
                loading: false,
                darkMode: false,
                user: {},
                status: {},
                config: {},
                recentLogs: [],
                chart: null,
                
                // Initialize
                async init() {
                    // Load Dark Mode preference
                    this.darkMode = localStorage.getItem('darkMode') === 'true';
                    this.applyDarkMode();
                    
                    // Check auth
                    const token = localStorage.getItem('access_token');
                    if (!token) {
                        window.location.href = '/login';
                        return;
                    }
                    
                    // Load data
                    await this.loadUser();
                    await this.refreshData();
                    
                    // Setup chart
                    this.setupChart();
                    
                    // Auto-refresh every 5 seconds
                    setInterval(() => this.refreshData(), 5000);
                },
                
                // Load user info
                async loadUser() {
                    try {
                        const response = await this.apiCall('/api/users/me');
                        this.user = response;
                    } catch (error) {
                        console.error('Failed to load user:', error);
                        this.logout();
                    }
                },
                
                // Refresh all data
                async refreshData() {
                    this.loading = true;
                    try {
                        await Promise.all([
                            this.loadStatus(),
                            this.loadConfig(),
                            this.loadLogs()
                        ]);
                    } finally {
                        this.loading = false;
                    }
                },
                
                // Load bot status
                async loadStatus() {
                    const data = await this.apiCall('/api/status');
                    this.status = data;
                },
                
                // Load config
                async loadConfig() {
                    const data = await this.apiCall('/api/config');
                    this.config = data;
                },
                
                // Save config
                async saveConfig() {
                    try {
                        await this.apiCall('/api/config', 'PUT', this.config);
                        alert('✅ تنظیمات ذخیره شد');
                    } catch (error) {
                        alert('❌ خطا در ذخیره');
                    }
                },
                
                // Load logs
                async loadLogs() {
                    const data = await this.apiCall('/api/logs?limit=10');
                    this.recentLogs = data.logs || [];
                },
                
                // API Call with JWT
                async apiCall(url, method = 'GET', body = null) {
                    const token = localStorage.getItem('access_token');
                    
                    const options = {
                        method,
                        headers: {
                            'Authorization': `Bearer ${token}`,
                            'Content-Type': 'application/json'
                        }
                    };
                    
                    if (body) {
                        options.body = JSON.stringify(body);
                    }
                    
                    const response = await fetch(url, options);
                    
                    if (response.status === 401) {
                        // Token expired
                        this.logout();
                        throw new Error('Unauthorized');
                    }
                    
                    return response.json();
                },
                
                // Setup Chart
                setupChart() {
                    const ctx = document.getElementById('performanceChart');
                    if (ctx) {
                        this.chart = new Chart(ctx, {
                            type: 'line',
                            data: {
                                labels: ['دوشنبه', 'سه‌شنبه', 'چهارشنبه', 'پنج‌شنبه', 'جمعه'],
                                datasets: [{
                                    label: 'سود/ضرر',
                                    data: [120, -50, 200, 150, 234],
                                    borderColor: 'rgb(147, 51, 234)',
                                    tension: 0.4
                                }]
                            },
                            options: {
                                responsive: true,
                                maintainAspectRatio: false
                            }
                        });
                    }
                },
                
                // Logout
                logout() {
                    localStorage.removeItem('access_token');
                    localStorage.removeItem('refresh_token');
                    window.location.href = '/login';
                },
                
                // Utils
                formatUptime(minutes) {
                    const hours = Math.floor(minutes / 60);
                    return hours.toFixed(1);
                },
                
                formatDate(timestamp) {
                    return new Date(timestamp).toLocaleString('fa-IR');
                },
                
                // Dark Mode
                toggleDarkMode() {
                    this.darkMode = !this.darkMode;
                    localStorage.setItem('darkMode', this.darkMode);
                    this.applyDarkMode();
                },
                
                applyDarkMode() {
                    if (this.darkMode) {
                        document.documentElement.classList.add('dark');
                    } else {
                        document.documentElement.classList.remove('dark');
                    }
                }
            }
        }
    </script>

        root.style.setProperty('--color-primary', theme.primary);
        root.style.setProperty('--color-secondary', theme.secondary);
        root.style.setProperty('--color-accent', theme.accent);
        root.style.setProperty('--color-background', theme.background);
        root.style.setProperty('--color-surface', theme.surface);
        root.style.setProperty('--color-text', theme.text);
        root.style.setProperty('--color-text-light', theme.textLight);
        root.style.setProperty('--color-success', theme.success);
        root.style.setProperty('--color-warning', theme.warning);
        root.style.setProperty('--color-error', theme.error);
        root.style.setProperty('--gradient', theme.gradient);
        
        // ذخیره
        this.currentTheme = themeName;
        localStorage.setItem('theme', themeName);
        
        // اعلان تغییر
        this.showNotification(`تم ${theme.name} اعمال شد`, 'success');
        
        // Dispatch Event
        window.dispatchEvent(new CustomEvent('themeChanged', { 
            detail: { theme: themeName } 
        }));
    }
    
    setupEventListeners() {
        // دکمه‌های تم
        document.querySelectorAll('[data-theme]').forEach(btn => {
            btn.addEventListener('click', (e) => {
                const themeName = e.currentTarget.dataset.theme;
                this.applyTheme(themeName);
                
                // به‌روزرسانی UI
                document.querySelectorAll('[data-theme]').forEach(b => {
                    b.classList.remove('active');
                });
                e.currentTarget.classList.add('active');
            });
        });
    }
    
    showNotification(message, type = 'info') {
        // استفاده از سیستم نوتیفیکیشن (بخش بعدی)
        if (window.NotificationSystem) {
            window.NotificationSystem.show(message, type);
        }
    }
    
    // دریافت تم فعلی
    getCurrentTheme() {
        return this.currentTheme;
    }
    
    // لیست تمام تم‌ها
    getAllThemes() {
        return Object.keys(THEMES).map(key => ({
            id: key,
            name: THEMES[key].name,
            primary: THEMES[key].primary
        }));
    }
}

/**
 * ====================================================================
 * فایل: charts.js
 * نمودارهای پیشرفته برای داشبورد
 * ====================================================================
 * شامل:
 * - Equity Curve (خط)
 * - Win Rate (Pie)
 * - Daily Profit (Bar)

class TradingCharts {
    constructor() {
        this.charts = {};
        this.defaultColors = {
            primary: '#7C3AED',
            success: '#10B981',
            danger: '#EF4444',
            warning: '#F59E0B',
            info: '#3B82F6'
        };
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 1. Equity Curve (نمودار خطی سرمایه)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createEquityCurve(canvasId, data) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        // نابود کردن نمودار قبلی
        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        this.charts[canvasId] = new Chart(ctx, {
            type: 'line',
            data: {
                labels: data.labels, // ['روز 1', 'روز 2', ...]
                datasets: [{
                    label: 'Equity',
                    data: data.values, // [10000, 10050, 10120, ...]
                    borderColor: this.defaultColors.primary,
                    backgroundColor: 'rgba(124, 58, 237, 0.1)',
                    borderWidth: 3,
                    fill: true,
                    tension: 0.4,
                    pointRadius: 0,
                    pointHoverRadius: 6,
                    pointHoverBackgroundColor: this.defaultColors.primary
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    intersect: false,
                    mode: 'index'
                },
                plugins: {
                    legend: {
                        display: false
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                return '$' + context.parsed.y.toLocaleString();
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: false,
                        ticks: {
                            callback: function(value) {
                                return '$' + value.toLocaleString();
                            }
                        }
                    }
                }
            }
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 2. Win Rate Pie Chart (نمودار دایره‌ای Win/Loss)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createWinRatePie(canvasId, wins, losses) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        const total = wins + losses;
        const winRate = total > 0 ? (wins / total * 100).toFixed(1) : 0;

        this.charts[canvasId] = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['برد', 'باخت'],
                datasets: [{
                    data: [wins, losses],
                    backgroundColor: [
                        this.defaultColors.success,
                        this.defaultColors.danger
                    ],
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                cutout: '70%',
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: {
                            padding: 20,
                            font: {
                                size: 14
                            }
                        }
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                const label = context.label || '';
                                const value = context.parsed;
                                const percentage = ((value / total) * 100).toFixed(1);
                                return `${label}: ${value} (${percentage}%)`;
                            }
                        }
                    }
                }
            },
            plugins: [{
                // نمایش Win Rate در وسط
                id: 'centerText',
                beforeDraw: function(chart) {
                    const ctx = chart.ctx;
                    const width = chart.width;
                    const height = chart.height;

                    ctx.restore();
                    const fontSize = (height / 114).toFixed(2);
                    ctx.font = fontSize + 'em sans-serif';
                    ctx.textBaseline = 'middle';
                    ctx.fillStyle = '#1F2937';

                    const text = `${winRate}%`;
                    const textX = Math.round((width - ctx.measureText(text).width) / 2);
                    const textY = height / 2;

                    ctx.fillText(text, textX, textY);
                    ctx.save();
                }
            }]
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 3. Daily Profit Bar Chart (نمودار میله‌ای سود روزانه)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createDailyProfitBar(canvasId, data) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        // رنگ بر اساس مثبت/منفی
        const backgroundColors = data.values.map(value => 
            value >= 0 ? this.defaultColors.success : this.defaultColors.danger
        );

        this.charts[canvasId] = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: data.labels, // ['دوشنبه', 'سه‌شنبه', ...]
                datasets: [{
                    label: 'سود/ضرر',
                    data: data.values, // [120, -50, 200, ...]
                    backgroundColor: backgroundColors,
                    borderRadius: 8,
                    borderSkipped: false
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        display: false
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                const value = context.parsed.y;
                                return value >= 0 ? `سود: $${value}` : `ضرر: $${Math.abs(value)}`;
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        ticks: {
                            callback: function(value) {
                                return '$' + value;
                            }
                        }
                    }
                }
            }
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 4. Drawdown Area Chart (نمودار Drawdown)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createDrawdownChart(canvasId, data) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        this.charts[canvasId] = new Chart(ctx, {
            type: 'line',
            data: {
                labels: data.labels,
                datasets: [{
                    label: 'Drawdown (%)',
                    data: data.values, // [0, -2, -3, -1, 0, ...]
                    borderColor: this.defaultColors.danger,
                    backgroundColor: 'rgba(239, 68, 68, 0.1)',
                    borderWidth: 2,
                    fill: true,
                    tension: 0.4,
                    pointRadius: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        display: false
                    },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                return context.parsed.y.toFixed(2) + '%';
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        max: 0,
                        ticks: {
                            callback: function(value) {
                                return value + '%';
                            }
                        }
                    }
                }
            }
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 5. Performance Gauge (عقربه عملکرد)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createPerformanceGauge(canvasId, value) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        // رنگ بر اساس مقدار
        let color;
        if (value >= 80) color = this.defaultColors.success;
        else if (value >= 60) color = this.defaultColors.info;
        else if (value >= 40) color = this.defaultColors.warning;
        else color = this.defaultColors.danger;

        this.charts[canvasId] = new Chart(ctx, {
            type: 'doughnut',
            data: {
                datasets: [{
                    data: [value, 100 - value],
                    backgroundColor: [color, '#E5E7EB'],
                    borderWidth: 0,
                    circumference: 180,
                    rotation: 270
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                cutout: '75%',
                plugins: {
                    legend: {
                        display: false
                    },
                    tooltip: {
                        enabled: false
                    }
                }
            },
            plugins: [{
                id: 'gaugeText',
                beforeDraw: function(chart) {
                    const ctx = chart.ctx;
                    const width = chart.width;
                    const height = chart.height;

                    ctx.restore();
                    const fontSize = (height / 80).toFixed(2);
                    ctx.font = 'bold ' + fontSize + 'em sans-serif';
                    ctx.textBaseline = 'middle';
                    ctx.fillStyle = color;

                    const text = value + '%';
                    const textX = Math.round((width - ctx.measureText(text).width) / 2);
                    const textY = height / 1.5;

                    ctx.fillText(text, textX, textY);
                    ctx.save();
                }
            }]
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 6. Multi-Line Comparison (مقایسه چند خط)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createComparisonChart(canvasId, datasets) {
        const ctx = document.getElementById(canvasId);
        if (!ctx) return null;

        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
        }

        const colors = [
            this.defaultColors.primary,
            this.defaultColors.success,
            this.defaultColors.info,
            this.defaultColors.warning
        ];

        const chartDatasets = datasets.map((dataset, index) => ({
            label: dataset.label,
            data: dataset.values,
            borderColor: colors[index % colors.length],
            backgroundColor: 'transparent',
            borderWidth: 2,
            tension: 0.4,
            pointRadius: 0,
            pointHoverRadius: 5
        }));

        this.charts[canvasId] = new Chart(ctx, {
            type: 'line',
            data: {
                labels: datasets[0].labels,
                datasets: chartDatasets
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    intersect: false,
                    mode: 'index'
                },
                plugins: {
                    legend: {
                        position: 'top',
                        align: 'end'
                    }
                }
            }
        });

        return this.charts[canvasId];
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // 7. Heatmap Calendar (تقویم گرمایی سود)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    createHeatmapCalendar(containerId, data) {
        // این یک نمونه ساده است
        // برای heatmap واقعی از کتابخانه‌هایی مثل d3.js استفاده کنید
        
        const container = document.getElementById(containerId);
        if (!container) return;

        let html = '<div class="grid grid-cols-7 gap-2">';
        
        data.forEach(day => {
            const color = day.profit > 0 ? 'bg-green-500' : 
                          day.profit < 0 ? 'bg-red-500' : 'bg-gray-200';
            const opacity = Math.min(Math.abs(day.profit) / 100, 1);
            
            html += `
                <div class="${color} rounded p-2 text-white text-xs text-center"
                     style="opacity: ${opacity}"
                     title="${day.date}: $${day.profit}">
                    ${day.day}
                </div>
            `;
        });
        
        html += '</div>';
        container.innerHTML = html;
    }

    // توابع کمکی

    destroyChart(canvasId) {
        if (this.charts[canvasId]) {
            this.charts[canvasId].destroy();
            delete this.charts[canvasId];
        }
    }

    destroyAll() {
        Object.keys(this.charts).forEach(key => {
            this.charts[key].destroy();
        });
        this.charts = {};
    }

    updateChart(canvasId, newData) {
        if (this.charts[canvasId]) {
            this.charts[canvasId].data = newData;
            this.charts[canvasId].update();
        }
    }

        function dashboardApp() {
            return {
                // State
                sidebarOpen: true,
                currentPage: 'overview',
                loading: false,
                user: {},
                status: {},
                config: {},
                recentLogs: [],
                chart: null,
                
                // Initialize
                async init() {
                    // Check auth
                    const token = localStorage.getItem('access_token');
                    if (!token) {
                        window.location.href = '/login';
                        return;
                    }
                    
                    // Load data
                    await this.loadUser();
                    await this.refreshData();
                    
                    // Setup chart
                    this.setupChart();
                    
                    // Auto-refresh every 5 seconds
                    setInterval(() => this.refreshData(), 5000);
                },
                
                // Load user info
                async loadUser() {
                    try {
                        const response = await this.apiCall('/api/users/me');
                        this.user = response;
                    } catch (error) {
                        console.error('Failed to load user:', error);
                        this.logout();
                    }
                },
                
                // Refresh all data
                async refreshData() {
                    this.loading = true;
                    try {
                        await Promise.all([
                            this.loadStatus(),
                            this.loadConfig(),
                            this.loadLogs()
                        ]);
                    } finally {
                        this.loading = false;
                    }
                },
                
                // Load bot status
                async loadStatus() {
                    const data = await this.apiCall('/api/status');
                    this.status = data;
                },
                
                // Load config
                async loadConfig() {
                    const data = await this.apiCall('/api/config');
                    this.config = data;
                },
                
                // Save config
                async saveConfig() {
                    try {
                        await this.apiCall('/api/config', 'PUT', this.config);
                        alert('✅ تنظیمات ذخیره شد');
                    } catch (error) {
                        alert('❌ خطا در ذخیره');
                    }
                },
                
                // Load logs
                async loadLogs() {
                    const data = await this.apiCall('/api/logs?limit=10');
                    this.recentLogs = data.logs || [];
                },
                
                // API Call with JWT
                async apiCall(url, method = 'GET', body = null) {
                    const token = localStorage.getItem('access_token');
                    
                    const options = {
                        method,
                        headers: {
                            'Authorization': `Bearer ${token}`,
                            'Content-Type': 'application/json'
                        }
                    };
                    
                    if (body) {
                        options.body = JSON.stringify(body);
                    }
                    
                    const response = await fetch(url, options);
                    
                    if (response.status === 401) {
                        // Token expired
                        this.logout();
                        throw new Error('Unauthorized');
                    }
                    
                    return response.json();
                },
                
                // Setup Chart
                setupChart() {
                    const ctx = document.getElementById('performanceChart');
                    if (ctx) {
                        this.chart = new Chart(ctx, {
                            type: 'line',
                            data: {
                                labels: ['دوشنبه', 'سه‌شنبه', 'چهارشنبه', 'پنج‌شنبه', 'جمعه'],
                                datasets: [{
                                    label: 'سود/ضرر',
                                    data: [120, -50, 200, 150, 234],
                                    borderColor: 'rgb(147, 51, 234)',
                                    tension: 0.4
                                }]
                            },
                            options: {
                                responsive: true,
                                maintainAspectRatio: false
                            }
                        });
                    }
                },
                
                // Logout
                logout() {
                    localStorage.removeItem('access_token');
                    localStorage.removeItem('refresh_token');
                    window.location.href = '/login';
                },
                
                // Utils
                formatUptime(minutes) {
                    const hours = Math.floor(minutes / 60);
                    return hours.toFixed(1);
                },
                
                formatDate(timestamp) {
                    return new Date(timestamp).toLocaleString('fa-IR');
                }
            }
        }
    </script>

// نصب Service Worker
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

self.addEventListener('install', (event) => {
  console.log('[SW] Installing Service Worker...');
  
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then((cache) => {
        console.log('[SW] Caching static assets');
        return cache.addAll(STATIC_ASSETS);
      })
      .then(() => {
        console.log('[SW] Service Worker installed successfully');
        return self.skipWaiting(); // فعال‌سازی فوری
      })
      .catch((error) => {
        console.error('[SW] Installation failed:', error);
      })
  );

// فعال‌سازی Service Worker
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

self.addEventListener('activate', (event) => {
  console.log('[SW] Activating Service Worker...');
  
  event.waitUntil(
    caches.keys()
      .then((cacheNames) => {
        // حذف cache های قدیمی
        return Promise.all(
          cacheNames
            .filter((name) => name !== CACHE_NAME && name !== DYNAMIC_CACHE)
            .map((name) => {
              console.log('[SW] Deleting old cache:', name);
              return caches.delete(name);
            })
        );
      })
      .then(() => {
        console.log('[SW] Service Worker activated');
        return self.clients.claim(); // کنترل فوری صفحات
      })
  );
});

// Fetch Strategy: Network First با Fallback به Cache
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

self.addEventListener('fetch', (event) => {
  const { request } = event;
  const url = new URL(request.url);
  
  // فقط درخواست‌های same-origin
  if (url.origin !== location.origin) {
    return;
  }
  
  // استراتژی بر اساس نوع درخواست
  if (request.url.includes('/api/')) {
    // API: Network Only (همیشه از سرور)
    event.respondWith(networkOnly(request));
  } else if (request.destination === 'document') {
    // HTML: Network First
    event.respondWith(networkFirst(request));
  } else {
    // Static Assets: Cache First
    event.respondWith(cacheFirst(request));
  }
});

/**
 * استراتژی Network First
 * ابتدا از شبکه، در صورت خطا از Cache
 */
async function networkFirst(request) {
  try {
    const networkResponse = await fetch(request);
    
    // Cache کردن برای دفعات بعد
    const cache = await caches.open(DYNAMIC_CACHE);
    cache.put(request, networkResponse.clone());
    
    return networkResponse;
  } catch (error) {
    console.log('[SW] Network failed, trying cache...');
    
    const cachedResponse = await caches.match(request);
    if (cachedResponse) {
      return cachedResponse;
    }
    
    // اگر هیچکدام نبود، صفحه Offline
    return caches.match('/offline.html');
  }
}

/**
 * استراتژی Cache First
 * ابتدا از Cache، در صورت نبود از شبکه
 */
async function cacheFirst(request) {
  const cachedResponse = await caches.match(request);
  
  if (cachedResponse) {
    // در پس‌زمینه به‌روزرسانی کن
    updateCache(request);
    return cachedResponse;
  }
  
  // اگر در Cache نبود، از شبکه
  try {
    const networkResponse = await fetch(request);
    
    // Cache کن
    const cache = await caches.open(DYNAMIC_CACHE);
    cache.put(request, networkResponse.clone());
    
    return networkResponse;
  } catch (error) {
    console.error('[SW] Fetch failed:', error);
    throw error;
  }
}

/**
 * استراتژی Network Only
 * فقط از شبکه (برای API)
 */
async function networkOnly(request) {
  return fetch(request);
}

/**
 * به‌روزرسانی Cache در پس‌زمینه
 */
async function updateCache(request) {
  try {
    const networkResponse = await fetch(request);
    const cache = await caches.open(DYNAMIC_CACHE);
    cache.put(request, networkResponse);
  } catch (error) {
    // در پس‌زمینه است، خطا مهم نیست
  }
}

/**
 * ====================================================================
 * سیستم نوتیفیکیشن پیشرفته
 * ====================================================================
 * Toast, Alert, Push Notification
 */

class NotificationSystem {
    constructor() {
        this.container = null;
        this.notifications = [];
        this.maxNotifications = 5;
        this.defaultDuration = 3000; // 3 ثانیه
        
        this.init();
    }
    
    init() {
        // ساخت Container
        this.createContainer();
        
        // درخواست مجوز Push Notification
        this.requestPushPermission();
    }
    
    createContainer() {
        // Container برای Toast ها
        this.container = document.createElement('div');
        this.container.id = 'notification-container';
        this.container.className = 'fixed top-4 left-4 z-[9999] space-y-3 pointer-events-none';
        this.container.style.cssText = 'max-width: 400px; width: 100%;';
        document.body.appendChild(this.container);
    }
    
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // Toast Notification
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    show(message, type = 'info', duration = this.defaultDuration) {
        // حذف اگر بیش از حد باشد
        if (this.notifications.length >= this.maxNotifications) {
            this.notifications[0].remove();
        }
        
        // ساخت Toast
        const toast = this.createToast(message, type);
        this.container.appendChild(toast);
        this.notifications.push(toast);
        
        // انیمیشن ورود
        setTimeout(() => {
            toast.style.transform = 'translateX(0)';
            toast.style.opacity = '1';
        }, 10);
        
        // حذف خودکار
        if (duration > 0) {
            setTimeout(() => {
                this.remove(toast);
            }, duration);
        }
        
        return toast;
    }
    
    createToast(message, type) {
        const toast = document.createElement('div');
        toast.className = 'notification-toast pointer-events-auto';
        
        // رنگ و آیکون بر اساس نوع
        const config = this.getTypeConfig(type);
        
        toast.style.cssText = `
            background: white;
            border-left: 4px solid ${config.color};
            border-radius: 0.75rem;
            box-shadow: 0 10px 25px rgba(0,0,0,0.1);
            padding: 1rem;
            display: flex;
            align-items: start;
            gap: 0.75rem;
            transform: translateX(100%);
            opacity: 0;
            transition: all 0.3s ease;
        `;
        
        toast.innerHTML = `
            <div style="
                width: 2.5rem;
                height: 2.5rem;
                border-radius: 50%;
                background: ${config.bgColor};
                display: flex;
                align-items: center;
                justify-content: center;
                flex-shrink: 0;
            ">
                <i class="${config.icon}" style="color: ${config.color}; font-size: 1.25rem;"></i>
            </div>
            <div style="flex: 1; padding-top: 0.25rem;">
                <p style="color: #1F2937; font-weight: 500; margin: 0;">${message}</p>
            </div>
            <button onclick="this.closest('.notification-toast').remove()" style="
                color: #9CA3AF;
                background: none;
                border: none;
                cursor: pointer;
                padding: 0;
                width: 1.5rem;
                height: 1.5rem;
            ">
                <i class="fas fa-times"></i>
            </button>
        `;
        
        return toast;
    }
    
    getTypeConfig(type) {
        const configs = {
            success: {
                color: '#10B981',
                bgColor: '#D1FAE5',
                icon: 'fas fa-check-circle'
            },
            error: {
                color: '#EF4444',
                bgColor: '#FEE2E2',
                icon: 'fas fa-exclamation-circle'
            },
            warning: {
                color: '#F59E0B',
                bgColor: '#FEF3C7',
                icon: 'fas fa-exclamation-triangle'
            },
            info: {
                color: '#3B82F6',
                bgColor: '#DBEAFE',
                icon: 'fas fa-info-circle'
            }
        };
        
        return configs[type] || configs.info;
    }
    
    remove(toast) {
        toast.style.transform = 'translateX(100%)';
        toast.style.opacity = '0';
        
        setTimeout(() => {
            toast.remove();
            const index = this.notifications.indexOf(toast);
            if (index > -1) {
                this.notifications.splice(index, 1);
            }
        }, 300);
    }
    
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // نوتیفیکیشن‌های خاص معاملاتی
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    tradeOpened(direction, price, volume) {
        const message = `🚀 معامله ${direction} باز شد\n💵 ${volume} لات @ ${price}`;
        this.show(message, 'success', 5000);
        
        // Push Notification
        this.sendPushNotification(
            'معامله باز شد',
            `${direction} ${volume} لات به قیمت ${price}`
        );
    }
    
    tradeClosed(profit, pips) {
        const type = profit >= 0 ? 'success' : 'error';
        const emoji = profit >= 0 ? '✅' : '❌';
        const message = `${emoji} معامله بسته شد\n💰 ${profit >= 0 ? '+' : ''}$${profit.toFixed(2)} (${pips >= 0 ? '+' : ''}${pips.toFixed(1)} پیپ)`;
        
        this.show(message, type, 5000);
        
        // Push
        this.sendPushNotification(
            profit >= 0 ? 'معامله سودده!' : 'معامله بسته شد',
            `سود/ضرر: $${profit.toFixed(2)}`
        );
    }
    
    partialClose(percent, profit) {
        const message = `💰 Partial Close\n${percent}% بسته شد - سود: $${profit.toFixed(2)}`;
        this.show(message, 'info', 4000);
    }
    
    breakEven(ticket) {
        const message = `🎯 Break Even فعال شد\nپوزیشن #${ticket}`;
        this.show(message, 'info', 4000);
    }
    
    dailyLossWarning(currentLoss, maxLoss) {
        const message = `⚠️ هشدار ضرر روزانه!\n${(currentLoss*100).toFixed(1)}% از ${(maxLoss*100).toFixed(0)}%`;
        this.show(message, 'warning', 0); // نمایش دائمی
        
        // Push
        this.sendPushNotification(
            '⚠️ هشدار ریسک',
            `ضرر روزانه به ${(currentLoss*100).toFixed(1)}% رسیده`
        );
    }
    
    modelRetrainNeeded(reason) {
        const message = `🔄 نیاز به Retrain\n${reason}`;
        this.show(message, 'warning', 0); // دائمی
    }
    
    connectionLost(service) {
        const message = `❌ قطع اتصال ${service}\nتلاش برای اتصال مجدد...`;
        this.show(message, 'error', 0);
    }
    
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // Alert Dialog (مودال)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    
    alert(title, message, type = 'info') {
        return new Promise((resolve) => {
            const modal = this.createAlertModal(title, message, type);
            document.body.appendChild(modal);
            
            // انیمیشن
            setTimeout(() => {
                modal.querySelector('.alert-dialog').style.transform = 'scale(1)';
                modal.querySelector('.alert-dialog').style.opacity = '1';
            }, 10);
            
            // دکمه OK
            modal.querySelector('.alert-ok-btn').addEventListener('click', () => {
                this.closeAlert(modal);
                resolve(true);
            });
        });
    }
    
    confirm(title, message) {
        return new Promise((resolve) => {
            const modal = this.createConfirmModal(title, message);
            document.body.appendChild(modal);
            
            // انیمیشن
            setTimeout(() => {
                modal.querySelector('.alert-dialog').style.transform = 'scale(1)';
                modal.querySelector('.alert-dialog').style.opacity = '1';
            }, 10);
            
            // دکمه‌ها
            modal.querySelector('.confirm-yes-btn').addEventListener('click', () => {
                this.closeAlert(modal);
                resolve(true);
            });
            
            modal.querySelector('.confirm-no-btn').addEventListener('click', () => {
                this.closeAlert(modal);
                resolve(false);
            });
        });
    }
    
    createAlertModal(title, message, type) {
        const modal = document.createElement('div');
        modal.className = 'alert-modal';
        modal.style.cssText = `

    async requestPushPermission() {
        if (!('Notification' in window)) {
            console.log('مرورگر از Push Notification پشتیبانی نمی‌کند');
            return;
        }
        
        if (Notification.permission === 'granted') {
            console.log('✅ Push Notification فعال است');
        } else if (Notification.permission !== 'denied') {
            const permission = await Notification.requestPermission();
            if (permission === 'granted') {
                console.log('✅ مجوز Push Notification داده شد');
            }
        }
    }
    
    sendPushNotification(title, body, icon = '/favicon.ico') {
        if (!('Notification' in window)) return;
        if (Notification.permission !== 'granted') return;
        
        const notification = new Notification(title, {
            body: body,
            icon: icon,
            badge: icon,
            tag: 'trading-bot',
            requireInteraction: false
        });
        
        // کلیک برای باز کردن داشبورد
        notification.onclick = () => {
            window.focus();
            notification.close();
        };
        
        // بستن خودکار بعد از 5 ثانیه
        setTimeout(() => notification.close(), 5000);
    }

class TestDashboardPerformance:
    """تست عملکرد داشبورد"""
    
    @pytest.fixture
    def auth_token(self):
        response = client.post(
            "/api/token",
            data={"username": "admin", "password": "admin123"}
        )
        return response.json()["access_token"]
    
    def test_api_response_time(self, auth_token):
        """زمان پاسخ API ها"""
        import time
        
        endpoints = [
            "/api/status",
            "/api/config",
            "/api/logs",
            "/api/logs/stats"
        ]
        
        for endpoint in endpoints:
            start = time.time()
            response = client.get(
                endpoint,
                headers={"Authorization": f"Bearer {auth_token}"}
            )
            duration = time.time() - start
            
            assert response.status_code == 200
            assert duration < 1.0, f"{endpoint} خیلی کند است: {duration:.2f}s"
    
    def test_concurrent_requests(self, auth_token):
        """درخواست‌های همزمان"""
        import threading
        
        results = []
        
        def make_request():
            response = client.get(
                "/api/status",
                headers={"Authorization": f"Bearer {auth_token}"}
            )
            results.append(response.status_code)
        
        # 10 درخواست همزمان
        threads = [threading.Thread(target=make_request) for _ in range(10)]
        
        for t in threads:
            t.start()
        
        for t in threads:
            t.join()
        
        # همه باید موفق باشند
        assert all(status == 200 for status in results)


class TestDashboardSecurity:
    """تست امنیت داشبورد"""
    
    def test_sql_injection_protection(self):
        """محافظت در برابر SQL Injection"""
        malicious_inputs = [
            "admin' OR '1'='1",
            "admin'; DROP TABLE users--",
            "admin' UNION SELECT * FROM logs--"
        ]
        
        for malicious in malicious_inputs:
            response = client.post(
                "/api/token",
                data={
                    "username": malicious,
                    "password": "anything"
                }
            )
            # باید 401 باشد نه 500
            assert response.status_code == 401
    
    def test_xss_protection(self, auth_token=None):
        """محافظت در برابر XSS"""
        if not auth_token:
            response = client.post(
                "/api/token",
                data={"username": "admin", "password": "admin123"}
            )
            auth_token = response.json()["access_token"]
        
        xss_payloads = [
            "<script>alert('XSS')</script>",
            "<img src=x onerror=alert('XSS')>",
            "javascript:alert('XSS')"
        ]
        
        for payload in xss_payloads:
            response = client.put(
                "/api/config",
                headers={"Authorization": f"Bearer {auth_token}"},
                json={"SYMBOL": payload}
            )
            # باید یا 200 با escape باشد یا 400 با validation error
            assert response.status_code in [200, 400, 422]
    
    def test_csrf_token_not_required_with_jwt(self):
        """با JWT نیاز به CSRF نیست"""
        # این تست فقط مفهومی است
        # در FastAPI با JWT، CSRF مشکل نیست
        response = client.post(
            "/api/token",
            data={"username": "admin", "password": "admin123"}
        )
        assert response.status_code == 200
    
    def test_rate_limiting_login_attempts(self):
