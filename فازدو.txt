@dataclass
class Trade:
    """نمایش یک معامله"""
    entry_time: datetime
    entry_price: float
    direction: str  # 'BUY' or 'SELL'
    volume: float
    sl: float
    tp: float
    entry_reason: str = ""
    
    # فیلدهای خروج
    exit_time: Optional[datetime] = None
    exit_price: Optional[float] = None
    exit_reason: Optional[str] = None
    
    # محاسبات مالی
    gross_profit: float = 0.0
    commission: float = 0.0
    net_profit: float = 0.0
    profit_pips: float = 0.0
    
    # مدیریت پوزیشن
    partial_closed: bool = False
    partial_volume: float = 0.0
    partial_profit: float = 0.0
    be_moved: bool = False
    
    # متریک‌ها
    duration_bars: int = 0
    mae: float = 0.0  # Maximum Adverse Excursion
    mfe: float = 0.0  # Maximum Favorable Excursion
    
    def __post_init__(self):
        """محاسبات اولیه"""
        self.ticket = int(datetime.now().timestamp() * 1000000)
    
    def calculate_profit(self, exit_price: float, commission_per_lot: float, 
                        contract_size: float = 100000):
        """محاسبه سود/ضرر"""
        if self.direction == 'BUY':
            self.gross_profit = (exit_price - self.entry_price) * self.volume * contract_size
            self.profit_pips = (exit_price - self.entry_price) * 10000
        else:  # SELL
            self.gross_profit = (self.entry_price - exit_price) * self.volume * contract_size
            self.profit_pips = (self.entry_price - exit_price) * 10000
        
        # کارمزد
        self.commission = commission_per_lot * self.volume * 2  # ورود + خروج
        
        # سود خالص
        self.net_profit = self.gross_profit - self.commission
        
        return self.net_profit


@dataclass
class BacktestResult:
    """نتایج Backtesting"""
    # اطلاعات کلی
    start_date: datetime
    end_date: datetime
    duration_days: int
    initial_capital: float
    final_capital: float
    
    # معاملات
    total_trades: int = 0
    winning_trades: int = 0
    losing_trades: int = 0
    break_even_trades: int = 0
    
    # سود/ضرر
    total_profit: float = 0.0
    total_loss: float = 0.0
    gross_profit: float = 0.0
    net_profit: float = 0.0
    total_commission: float = 0.0
    
    # متریک‌های عملکرد
    win_rate: float = 0.0
    profit_factor: float = 0.0
    average_win: float = 0.0
    average_loss: float = 0.0
    largest_win: float = 0.0
    largest_loss: float = 0.0
    
    # ریسک
    max_drawdown: float = 0.0
    max_drawdown_pct: float = 0.0
    sharpe_ratio: float = 0.0
    sortino_ratio: float = 0.0
    
    # آمار معامله
    avg_trade_duration: float = 0.0
    avg_trade_profit: float = 0.0
    total_pips: float = 0.0
    
    # لیست معاملات
    trades: List[Trade] = field(default_factory=list)
    equity_curve: List[float] = field(default_factory=list)
    
    def calculate_metrics(self):
        """محاسبه تمام متریک‌ها"""
        if self.total_trades == 0:
            return
        
        # Win Rate
        self.win_rate = self.winning_trades / self.total_trades
        
        # Profit Factor
        if self.total_loss != 0:
            self.profit_factor = abs(self.total_profit / self.total_loss)
        else:
            self.profit_factor = float('inf') if self.total_profit > 0 else 0
        
        # Average Win/Loss
        if self.winning_trades > 0:
            self.average_win = self.total_profit / self.winning_trades
        if self.losing_trades > 0:
            self.average_loss = self.total_loss / self.losing_trades
        
        # Largest Win/Loss
        if self.trades:
            profits = [t.net_profit for t in self.trades if t.net_profit > 0]
            losses = [t.net_profit for t in self.trades if t.net_profit < 0]
            
            self.largest_win = max(profits) if profits else 0
            self.largest_loss = min(losses) if losses else 0
        
        # Average Trade
        self.avg_trade_profit = self.net_profit / self.total_trades
        
        # Duration
        durations = [t.duration_bars for t in self.trades if t.duration_bars > 0]
        self.avg_trade_duration = np.mean(durations) if durations else 0
        
        # Pips
        self.total_pips = sum([t.profit_pips for t in self.trades])
        
        # Sharpe Ratio (ساده‌شده)
        if self.trades:
            returns = [t.net_profit for t in self.trades]
            if np.std(returns) > 0:
                self.sharpe_ratio = np.mean(returns) / np.std(returns) * np.sqrt(252)
        
        # Return
        if self.initial_capital > 0:
            total_return_pct = (self.final_capital - self.initial_capital) / self.initial_capital * 100
        
    def to_dict(self):
        """تبدیل به Dictionary"""
        return {
            'summary': {
                'period': f"{self.start_date.date()} to {self.end_date.date()}",
                'duration_days': self.duration_days,
                'initial_capital': self.initial_capital,
                'final_capital': self.final_capital,
                'net_profit': self.net_profit,
                'return_pct': ((self.final_capital - self.initial_capital) / self.initial_capital * 100),
            },
            'trades': {
                'total': self.total_trades,
                'winning': self.winning_trades,
                'losing': self.losing_trades,
                'break_even': self.break_even_trades,
            },
            'performance': {
                'win_rate': self.win_rate,
                'profit_factor': self.profit_factor,
                'sharpe_ratio': self.sharpe_ratio,
                'max_drawdown_pct': self.max_drawdown_pct,
            },
            'trade_stats': {
                'average_win': self.average_win,
                'average_loss': self.average_loss,
                'largest_win': self.largest_win,
                'largest_loss': self.largest_loss,
                'avg_duration_bars': self.avg_trade_duration,
                'total_pips': self.total_pips,
            },
            'costs': {
                'total_commission': self.total_commission,
                'commission_per_trade': self.total_commission / self.total_trades if self.total_trades > 0 else 0,
            }
        }


class BacktestEngine:
    """موتور اصلی Backtesting"""
    
    def __init__(self, config: Dict):
        """
        Args:
            config: تنظیمات backtesting
                - initial_capital: سرمایه اولیه
                - commission_per_lot: کارمزد به ازای هر لات
                - spread_pips: اسپرد به پیپ
                - slippage_pips: لغزش به پیپ
                - contract_size: اندازه قرارداد
        """
        self.config = config
        self.trades: List[Trade] = []
        self.open_trades: List[Trade] = []
        self.equity = config['initial_capital']
        self.equity_curve = [self.equity]
        self.balance = config['initial_capital']
        
        # تنظیمات پیش‌فرض
        self.commission_per_lot = config.get('commission_per_lot', 7.0)
        self.spread_pips = config.get('spread_pips', 1.0)
        self.slippage_pips = config.get('slippage_pips', 0.5)
        self.contract_size = config.get('contract_size', 100000)
        
        # آمار
        self.total_signals = 0
        self.signals_taken = 0
        self.signals_filtered = 0
        
    def add_spread_slippage(self, price: float, direction: str) -> float:
        """اضافه کردن اسپرد و slippage به قیمت"""
        pip = 0.0001
        
        if direction == 'BUY':
            # خرید: قیمت Ask + slippage
            return price + (self.spread_pips + self.slippage_pips) * pip
        else:  # SELL
            # فروش: قیمت Bid - slippage
            return price - self.slippage_pips * pip
    
    def can_open_trade(self, volume: float, max_positions: int = 2) -> bool:
        """بررسی امکان باز کردن معامله"""
        # بررسی تعداد پوزیشن‌های باز
        if len(self.open_trades) >= max_positions:
            return False
        
        # بررسی مارجین کافی (ساده‌شده)
        required_margin = volume * self.contract_size * 0.01  # 1% margin
        if self.balance < required_margin:
            return False
        
        return True
    
    def open_trade(self, bar: pd.Series, signal: Dict) -> Optional[Trade]:
        """
        باز کردن معامله جدید
        
        Args:
            bar: کندل فعلی
            signal: سیگنال شامل:
                - direction: 'BUY' or 'SELL'
                - volume: حجم
                - sl: stop loss
                - tp: take profit
                - reason: دلیل ورود
        """
        self.total_signals += 1
        
        # بررسی امکان معامله
        if not self.can_open_trade(signal['volume']):
            self.signals_filtered += 1
            return None
        
        # قیمت واقعی (با اسپرد و slippage)
        entry_price = self.add_spread_slippage(bar['close'], signal['direction'])
        
        # ساخت Trade
        trade = Trade(
            entry_time=bar['time'],
            entry_price=entry_price,
            direction=signal['direction'],
            volume=signal['volume'],
            sl=signal['sl'],
            tp=signal['tp'],
            entry_reason=signal.get('reason', '')
        )
        
        self.open_trades.append(trade)
        self.signals_taken += 1
        
        return trade
    
    def update_open_trades(self, bar: pd.Series, bar_index: int):
        """
        به‌روزرسانی معاملات باز
        - بررسی SL/TP
        - محاسبه MAE/MFE
        """
        for trade in self.open_trades[:]:  # کپی لیست
            # محاسبه سود فعلی
            if trade.direction == 'BUY':
                current_profit = (bar['close'] - trade.entry_price) * trade.volume * self.contract_size
                # MAE/MFE
                if bar['low'] < trade.entry_price:
                    trade.mae = min(trade.mae, (bar['low'] - trade.entry_price) * 10000)
                if bar['high'] > trade.entry_price:
                    trade.mfe = max(trade.mfe, (bar['high'] - trade.entry_price) * 10000)
            else:  # SELL
                current_profit = (trade.entry_price - bar['close']) * trade.volume * self.contract_size
                # MAE/MFE
                if bar['high'] > trade.entry_price:
                    trade.mae = min(trade.mae, (trade.entry_price - bar['high']) * 10000)
                if bar['low'] < trade.entry_price:
                    trade.mfe = max(trade.mfe, (trade.entry_price - bar['low']) * 10000)
            
            # بررسی SL
            sl_hit = False
            if trade.direction == 'BUY' and bar['low'] <= trade.sl:
                sl_hit = True
                exit_price = trade.sl
            elif trade.direction == 'SELL' and bar['high'] >= trade.sl:
                sl_hit = True
                exit_price = trade.sl
            
            # بررسی TP
            tp_hit = False
            if trade.direction == 'BUY' and bar['high'] >= trade.tp:
                tp_hit = True
                exit_price = trade.tp
            elif trade.direction == 'SELL' and bar['low'] <= trade.tp:
                tp_hit = True
                exit_price = trade.tp
            
            # بستن معامله
            if sl_hit or tp_hit:
                # اضافه کردن slippage
                exit_price = self.add_spread_slippage(
                    exit_price, 
                    'SELL' if trade.direction == 'BUY' else 'BUY'
                )
                
                self.close_trade(
                    trade, 
                    bar['time'], 
                    exit_price,
                    'SL Hit' if sl_hit else 'TP Hit',
                    bar_index
                )
    
    def close_trade(self, trade: Trade, exit_time: datetime, exit_price: float, 
                   reason: str, bar_index: int):
        """بستن معامله"""
        trade.exit_time = exit_time
        trade.exit_price = exit_price
        trade.exit_reason = reason
        
        # محاسبه سود
        trade.calculate_profit(exit_price, self.commission_per_lot, self.contract_size)
        
        # محاسبه مدت
        # بعداً در post-processing
        
        # به‌روزرسانی موجودی
        self.balance += trade.net_profit
        self.equity = self.balance
        
        # حذف از لیست باز
        if trade in self.open_trades:
            self.open_trades.remove(trade)
        
        # اضافه به لیست بسته
        self.trades.append(trade)
        
        # به‌روزرسانی equity curve
        self.equity_curve.append(self.equity)
    
    def calculate_drawdown(self) -> Tuple[float, float]:
        """محاسبه Maximum Drawdown"""
        equity_array = np.array(self.equity_curve)
        
        # محاسبه Running Maximum
        running_max = np.maximum.accumulate(equity_array)
        
        # Drawdown در هر نقطه
        drawdown = equity_array - running_max
        
        # Max Drawdown
        max_dd = np.min(drawdown)
        max_dd_pct = (max_dd / running_max[np.argmin(drawdown)]) * 100 if running_max[np.argmin(drawdown)] > 0 else 0
        
        return max_dd, max_dd_pct
    
    def generate_result(self, df: pd.DataFrame) -> BacktestResult:
        """تولید نتیجه نهایی"""
        result = BacktestResult(
            start_date=df['time'].iloc[0],
            end_date=df['time'].iloc[-1],
            duration_days=(df['time'].iloc[-1] - df['time'].iloc[0]).days,
            initial_capital=self.config['initial_capital'],
            final_capital=self.balance,
            trades=self.trades,
            equity_curve=self.equity_curve
        )
        
        # طبقه‌بندی معاملات
        for trade in self.trades:
            if trade.net_profit > 0.01:
                result.winning_trades += 1
                result.total_profit += trade.net_profit
            elif trade.net_profit < -0.01:
                result.losing_trades += 1
                result.total_loss += trade.net_profit
            else:
                result.break_even_trades += 1
        
        result.total_trades = len(self.trades)
        result.gross_profit = sum([t.gross_profit for t in self.trades])
        result.net_profit = sum([t.net_profit for t in self.trades])
        result.total_commission = sum([t.commission for t in self.trades])
        
        # Drawdown
        result.max_drawdown, result.max_drawdown_pct = self.calculate_drawdown()
        
        # محاسبه تمام متریک‌ها
        result.calculate_metrics()
        
        return result
    
    def print_progress(self, current: int, total: int, bar_time: datetime):
        """نمایش پیشرفت"""
        progress = (current / total) * 100
        bar_length = 40
        filled = int(bar_length * current / total)
        bar = '█' * filled + '░' * (bar_length - filled)
        
        print(f"\r[{bar}] {progress:.1f}% | {bar_time.date()} | معاملات: {len(self.trades)} | Equity: ${self.equity:,.0f}", 
              end='', flush=True)

    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # گام 3: راه‌اندازی Backtest Engine
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    print("\n" + "─"*70)
    print("⚙️ راه‌اندازی موتور Backtesting...")
    print("─"*70)
    
    engine_config = {
        'initial_capital': initial_capital,
        'commission_per_lot': commission,
        'spread_pips': spread,
        'slippage_pips': 0.5,
    }
    
    engine = BacktestEngine(engine_config)
    risk_manager = RiskManager()
    
    print("✅ موتور آماده است")
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # گام 4: حلقه اصلی Backtesting
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    print("\n" + "─"*70)
    print("🔄 شروع Backtesting...")
    print("─"*70)
    
    lookback = Config.LOOKBACK_PERIOD
    
    # شروع از lookback به بعد
    for i in range(lookback, len(df)):
        current_bar = df.iloc[i]
        
        # نمایش پیشرفت
        if i % 100 == 0:
            engine.print_progress(i - lookback, len(df) - lookback, current_bar['time'])
        
        # به‌روزرسانی معاملات باز
        engine.update_open_trades(current_bar, i)
        
        # اگر معامله باز داریم، ادامه
        if len(engine.open_trades) >= Config.MAX_POSITIONS:
            continue
        
        # آماده‌سازی داده برای پیش‌بینی
        features_window = df.iloc[i-lookback:i][model.features].values
        X_scaled = model.scaler.transform(features_window)
        X = X_scaled.reshape(1, lookback, -1)
        
        # پیش‌بینی
        prediction = model.predict(X)
        
        # اندیکاتورها
        current_atr = df.iloc[i]['ATR']
        avg_atr = df.iloc[i-20:i]['ATR'].mean()
        current_adx = df.iloc[i]['ADX']
        current_rsi = df.iloc[i]['RSI']
        
        # Mock tick برای فیلترها
        class MockTick:
            def __init__(self, close, spread_pips):
                pip = 0.0001
                self.bid = close
                self.ask = close + spread_pips * pip
        
        tick = MockTick(current_bar['close'], spread)
        
        # بررسی فیلترها (اگر فعال باشد)
        if use_filters:
            passed, reason, details = risk_manager.check_market_conditions(
                df.iloc[i-50:i+1], tick
            )
            
            if not passed:
                continue
        
        # تصمیم‌گیری
        signal = None
        
        # سیگنال خرید
        if prediction > Config.BUY_THRESHOLD and current_adx > Config.ADX_THRESHOLD:
            # محاسبه SL/TP
            sl, tp, sl_dist, tp_dist = risk_manager.calculate_dynamic_sl_tp(
                entry_price=current_bar['close'],
                direction='BUY',
                current_atr=current_atr,
                avg_atr=avg_atr,
                adx=current_adx
            )
            
            # محاسبه حجم
            volume = risk_manager.calculate_volume(current_bar['close'], sl)
            
            signal = {
                'direction': 'BUY',
                'volume': volume,
                'sl': sl,
                'tp': tp,
                'reason': f"ML:{prediction:.3f} ADX:{current_adx:.1f}"
            }
        
        # سیگنال فروش
        elif prediction < Config.SELL_THRESHOLD and current_adx > Config.ADX_THRESHOLD:
            # محاسبه SL/TP
            sl, tp, sl_dist, tp_dist = risk_manager.calculate_dynamic_sl_tp(
                entry_price=current_bar['close'],
                direction='SELL',
                current_atr=current_atr,
                avg_atr=avg_atr,
                adx=current_adx
            )
            
            # محاسبه حجم
            volume = risk_manager.calculate_volume(current_bar['close'], sl)
            
            signal = {
                'direction': 'SELL',
                'volume': volume,
                'sl': sl,
                'tp': tp,
                'reason': f"ML:{prediction:.3f} ADX:{current_adx:.1f}"
            }
        
        # باز کردن معامله
        if signal:
            trade = engine.open_trade(current_bar, signal)
    
    # پایان حلقه
    print()  # خط جدید بعد از progress bar
    
    # بستن معاملات باز
    if engine.open_trades:
        print(f"\n⚠️ بستن {len(engine.open_trades)} معامله باز...")
        for trade in engine.open_trades[:]:
            engine.close_trade(
                trade, 
                df.iloc[-1]['time'],
                df.iloc[-1]['close'],
                'End of Backtest',
                len(df) - 1
            )
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # گام 5: تولید نتایج
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    print("\n" + "─"*70)
    print("📊 تولید گزارش...")
    print("─"*70)
    
    # محاسبه duration برای هر trade
    for trade in engine.trades:
        if trade.exit_time and trade.entry_time:
            # پیدا کردن index
            entry_idx = df[df['time'] == trade.entry_time].index[0]
            exit_idx = df[df['time'] == trade.exit_time].index[0]
            trade.duration_bars = exit_idx - entry_idx
    
    result = engine.generate_result(df)
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # نمایش نتایج
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    print_results(result, engine)
    
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    # ذخیره نتایج
    # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    save_results(result, engine, symbol, timeframe)
    
    return result


def print_results(result: BacktestResult, engine: BacktestEngine):
    """چاپ نتایج"""
    
    print("\n" + "="*70)
    print("📊 نتایج Backtesting")
    print("="*70)
    
    # خلاصه
    print(f"\n🗓️  دوره: {result.start_date.date()} تا {result.end_date.date()} ({result.duration_days} روز)")
    print(f"💰 سرمایه اولیه: ${result.initial_capital:,.0f}")
    print(f"💵 سرمایه نهایی: ${result.final_capital:,.0f}")
    
    # سود/ضرر
    total_return = ((result.final_capital - result.initial_capital) / result.initial_capital) * 100
    color = '🟢' if total_return > 0 else '🔴'
    print(f"{color} سود/ضرر: ${result.net_profit:,.2f} ({total_return:+.2f}%)")
    
    # معاملات
    print(f"\n📊 معاملات:")
    print(f"   کل: {result.total_trades}")
    print(f"   ✅ برنده: {result.winning_trades} ({result.win_rate*100:.1f}%)")
    print(f"   ❌ بازنده: {result.losing_trades}")
    print(f"   ➖ صفر: {result.break_even_trades}")
    
    # متریک‌های عملکرد
    print(f"\n📈 عملکرد:")
    print(f"   Profit Factor: {result.profit_factor:.2f}")
    print(f"   Sharpe Ratio: {result.sharpe_ratio:.2f}")
    print(f"   Max Drawdown: ${result.max_drawdown:,.2f} ({result.max_drawdown_pct:.2f}%)")
    
    # آمار معاملات
    print(f"\n📊 آمار:")
    print(f"   میانگین برد: ${result.average_win:,.2f}")
    print(f"   میانگین باخت: ${result.average_loss:,.2f}")
    print(f"   بزرگترین برد: ${result.largest_win:,.2f}")
    print(f"   بزرگترین باخت: ${result.largest_loss:,.2f}")
    print(f"   میانگین مدت: {result.avg_trade_duration:.1f} کندل")
    print(f"   کل پیپ: {result.total_pips:,.1f}")
    
    # هزینه‌ها
    print(f"\n💸 هزینه‌ها:")
    print(f"   کل کارمزد: ${result.total_commission:,.2f}")
    print(f"   کارمزد هر معامله: ${result.total_commission / result.total_trades:.2f}")
    
    # سیگنال‌ها
    print(f"\n🎯 سیگنال‌ها:")
    print(f"   کل سیگنال: {engine.total_signals}")
    print(f"   معاملات انجام شده: {engine.signals_taken}")
    print(f"   فیلتر شده: {engine.signals_filtered}")
    
    # ارزیابی کلی
    print(f"\n{'='*70}")
    
    # معیارهای موفقیت
    is_profitable = result.net_profit > 0
    is_good_wr = result.win_rate > 0.45
    is_good_pf = result.profit_factor > 1.5
    is_good_sharpe = result.sharpe_ratio > 1.0
    is_low_dd = result.max_drawdown_pct < 20
    
    score = sum([is_profitable, is_good_wr, is_good_pf, is_good_sharpe, is_low_dd])
    
    if score >= 4:
        print("✅ ارزیابی: عالی - استراتژی پتانسیل بالایی دارد")
    elif score >= 3:
        print("✅ ارزیابی: خوب - استراتژی قابل استفاده است")
    elif score >= 2:
        print("⚠️ ارزیابی: متوسط - نیاز به بهبود دارد")
    else:
        print("❌ ارزیابی: ضعیف - استراتژی نیاز به بازنگری دارد")
    
    print("="*70)


def save_results(result: BacktestResult, engine: BacktestEngine, symbol: str, timeframe: str):
    """ذخیره نتایج"""
    
    # ساخت پوشه
    os.makedirs('backtest_results', exist_ok=True)
    
    # نام فایل
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    filename = f"backtest_results/{symbol}_{timeframe}_{timestamp}"
    
    # ذخیره JSON
    result_dict = result.to_dict()
    result_dict['config'] = {
        'symbol': symbol,
        'timeframe': timeframe,
        'total_signals': engine.total_signals,
        'signals_taken': engine.signals_taken,
        'signals_filtered': engine.signals_filtered,
    }
    
    with open(f"{filename}.json", 'w') as f:
        json.dump(result_dict, f, indent=2)
    
    print(f"\n💾 نتایج ذخیره شد: {filename}.json")
    
    # ذخیره CSV معاملات
    if result.trades:
        trades_data = []
        for trade in result.trades:
            trades_data.append({
                'entry_time': trade.entry_time,
                'exit_time': trade.exit_time,
                'direction': trade.direction,
                'entry_price': trade.entry_price,
                'exit_price': trade.exit_price,
                'volume': trade.volume,
                'sl': trade.sl,
                'tp': trade.tp,
                'net_profit': trade.net_profit,
                'profit_pips': trade.profit_pips,
                'duration_bars': trade.duration_bars,
                'exit_reason': trade.exit_reason,
            })
        
        trades_df = pd.DataFrame(trades_data)
        trades_df.to_csv(f"{filename}_trades.csv", index=False)
        print(f"💾 معاملات ذخیره شد: {filename}_trades.csv")

import matplotlib.pyplot as plt
import matplotlib.dates as mdates
from matplotlib.backends.backend_pdf import PdfPages
import pandas as pd
import numpy as np
from datetime import datetime
from typing import List, Dict
import seaborn as sns

# تنظیمات فونت فارسی
plt.rcParams['font.family'] = 'DejaVu Sans'
sns.set_style("whitegrid")


class BacktestVisualizer:
    """تولید نمودارها و گزارش بصری"""
    
    def __init__(self, result, engine=None):
        """
        Args:
            result: BacktestResult object
            engine: BacktestEngine object (optional)
        """
        self.result = result
        self.engine = engine
        self.figures = []
    
    def plot_equity_curve(self, save_path: str = None):
        """رسم Equity Curve"""
        fig, ax = plt.subplots(figsize=(14, 6))
        
        # داده‌ها
        equity = self.result.equity_curve
        x = range(len(equity))
        
        # رسم Equity
        ax.plot(x, equity, linewidth=2, color='#2196F3', label='Equity')
        ax.fill_between(x, equity, alpha=0.3, color='#2196F3')
        
        # خط سرمایه اولیه
        ax.axhline(y=self.result.initial_capital, 
                   color='gray', linestyle='--', linewidth=1, 
                   label='Initial Capital')
        
        # محاسبه Running Maximum
        running_max = np.maximum.accumulate(equity)
        
        # رسم Running Max
        ax.plot(x, running_max, linewidth=1, color='green', 
                linestyle='--', alpha=0.5, label='Peak')
        
        # محاسبه Drawdown
        drawdown = np.array(equity) - running_max
        
        # رنگ Drawdown
        ax.fill_between(x, equity, running_max, 
                        where=(drawdown < 0), 
                        color='red', alpha=0.3, 
                        label='Drawdown')
        
        # تنظیمات
        ax.set_xlabel('Trade Number', fontsize=11, fontweight='bold')
        ax.set_ylabel('Equity ($)', fontsize=11, fontweight='bold')
        ax.set_title('Equity Curve & Drawdown', fontsize=14, fontweight='bold')
        ax.legend(loc='best', fontsize=10)
        ax.grid(True, alpha=0.3)
        
        # فرمت محور Y
        ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda x, p: f'${x:,.0f}'))
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✅ Equity Curve ذخیره شد: {save_path}")
        
        self.figures.append(fig)
        return fig
    
    def plot_drawdown(self, save_path: str = None):
        """رسم Drawdown جداگانه"""
        fig, ax = plt.subplots(figsize=(14, 5))
        
        # محاسبه Drawdown
        equity = np.array(self.result.equity_curve)
        running_max = np.maximum.accumulate(equity)
        drawdown = equity - running_max
        drawdown_pct = (drawdown / running_max) * 100
        
        x = range(len(drawdown_pct))
        
        # رسم
        ax.fill_between(x, 0, drawdown_pct, color='red', alpha=0.5)
        ax.plot(x, drawdown_pct, color='darkred', linewidth=1.5)
        
        # خط Max Drawdown
        ax.axhline(y=self.result.max_drawdown_pct, 
                   color='red', linestyle='--', linewidth=2,
                   label=f'Max DD: {self.result.max_drawdown_pct:.2f}%')
        
        # تنظیمات
        ax.set_xlabel('Trade Number', fontsize=11, fontweight='bold')
        ax.set_ylabel('Drawdown (%)', fontsize=11, fontweight='bold')
        ax.set_title('Drawdown Chart', fontsize=14, fontweight='bold')
        ax.legend(loc='best', fontsize=10)
        ax.grid(True, alpha=0.3)
        
        # فرمت محور Y
        ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda x, p: f'{x:.1f}%'))
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✅ Drawdown Chart ذخیره شد: {save_path}")
        
        self.figures.append(fig)
        return fig
    
    def plot_trade_analysis(self, save_path: str = None):
        """آنالیز معاملات"""
        fig, axes = plt.subplots(2, 2, figsize=(14, 10))
        
        # داده‌های معاملات
        trades = self.result.trades
        profits = [t.net_profit for t in trades]
        durations = [t.duration_bars for t in trades if t.duration_bars > 0]
        pips = [t.profit_pips for t in trades]
        
        # 1. هیستوگرام سود/ضرر
        ax1 = axes[0, 0]
        colors = ['green' if p > 0 else 'red' for p in profits]
        ax1.bar(range(len(profits)), profits, color=colors, alpha=0.6, edgecolor='black')
        ax1.axhline(y=0, color='black', linewidth=1)
        ax1.set_xlabel('Trade Number', fontweight='bold')
        ax1.set_ylabel('Profit ($)', fontweight='bold')
        ax1.set_title('Individual Trade Profit/Loss', fontweight='bold')
        ax1.grid(True, alpha=0.3)
        
        # 2. توزیع سود/ضرر
        ax2 = axes[0, 1]
        ax2.hist(profits, bins=30, color='steelblue', alpha=0.7, edgecolor='black')
        ax2.axvline(x=0, color='red', linestyle='--', linewidth=2)
        ax2.axvline(x=np.mean(profits), color='green', linestyle='--', linewidth=2, 
                    label=f'Mean: ${np.mean(profits):.2f}')
        ax2.set_xlabel('Profit ($)', fontweight='bold')
        ax2.set_ylabel('Frequency', fontweight='bold')
        ax2.set_title('Profit Distribution', fontweight='bold')
        ax2.legend()
        ax2.grid(True, alpha=0.3)
        
        # 3. مدت معاملات
        ax3 = axes[1, 0]
        if durations:
            ax3.hist(durations, bins=20, color='orange', alpha=0.7, edgecolor='black')
            ax3.axvline(x=np.mean(durations), color='red', linestyle='--', linewidth=2,
                       label=f'Mean: {np.mean(durations):.1f} bars')
            ax3.set_xlabel('Duration (bars)', fontweight='bold')
            ax3.set_ylabel('Frequency', fontweight='bold')
            ax3.set_title('Trade Duration Distribution', fontweight='bold')
            ax3.legend()
            ax3.grid(True, alpha=0.3)
        
        # 4. Cumulative Profit
        ax4 = axes[1, 1]
        cumulative = np.cumsum(profits)
        ax4.plot(cumulative, linewidth=2, color='purple')
        ax4.fill_between(range(len(cumulative)), cumulative, alpha=0.3, color='purple')
        ax4.axhline(y=0, color='black', linewidth=1)
        ax4.set_xlabel('Trade Number', fontweight='bold')
        ax4.set_ylabel('Cumulative Profit ($)', fontweight='bold')
        ax4.set_title('Cumulative Profit', fontweight='bold')
        ax4.grid(True, alpha=0.3)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✅ Trade Analysis ذخیره شد: {save_path}")
        
        self.figures.append(fig)
        return fig
    
    def plot_monthly_returns(self, save_path: str = None):
        """بازدهی ماهانه"""
        fig, ax = plt.subplots(figsize=(14, 6))
        
        # گروه‌بندی معاملات بر اساس ماه
        trades_df = pd.DataFrame([{
            'date': t.exit_time,

        if len(trades_df) == 0:

            return None
        
        trades_df['month'] = pd.to_datetime(trades_df['date']).dt.to_period('M')
        monthly = trades_df.groupby('month')['profit'].sum()
        
        # رنگ‌بندی
        colors = ['green' if v > 0 else 'red' for v in monthly.values]
        
        # رسم
        monthly.plot(kind='bar', ax=ax, color=colors, alpha=0.7, edgecolor='black')
        ax.axhline(y=0, color='black', linewidth=1)
        
        # تنظیمات
        ax.set_xlabel('Month', fontsize=11, fontweight='bold')
        ax.set_ylabel('Profit ($)', fontsize=11, fontweight='bold')
        ax.set_title('Monthly Returns', fontsize=14, fontweight='bold')
        ax.grid(True, alpha=0.3)
        ax.tick_params(axis='x', rotation=45)
        
        # فرمت
        ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda x, p: f'${x:,.0f}'))
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')

    def plot_win_loss_analysis(self, save_path: str = None):

        trades = self.result.trades

        losses = [t.net_profit for t in trades if t.net_profit < 0]
        
        # 1. نمودار دایره‌ای Win/Loss
        ax1 = axes[0]
        sizes = [len(wins), len(losses)]
        labels = [f'Wins\n{len(wins)}', f'Losses\n{len(losses)}']
        colors = ['#4CAF50', '#F44336']
        explode = (0.05, 0)
        
        ax1.pie(sizes, labels=labels, colors=colors, explode=explode,
                autopct='%1.1f%%', startangle=90, textprops={'fontsize': 11, 'fontweight': 'bold'})
        ax1.set_title('Win/Loss Ratio', fontweight='bold')
        
        # 2. مقایسه میانگین
        ax2 = axes[1]
        avg_win = np.mean(wins) if wins else 0
        avg_loss = np.mean(losses) if losses else 0
        
        bars = ax2.bar(['Avg Win', 'Avg Loss'], 
                       [avg_win, abs(avg_loss)],
                       color=['green', 'red'], alpha=0.7, edgecolor='black')

        ax2.set_title('Average Win vs Loss', fontweight='bold')
        ax2.grid(True, alpha=0.3, axis='y')

        ax3 = axes[2]
        data = [wins, [abs(l) for l in losses]]

            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✅ Win/Loss Analysis ذخیره شد: {save_path}")

        return fig

"""
====================================================================
فایل: monte_carlo.py
شبیه‌سازی Monte Carlo برای ارزیابی ریسک
====================================================================
"""

import numpy as np
import matplotlib.pyplot as plt
from typing import List, Dict
from dataclasses import dataclass
import pandas as pd


@dataclass
class MonteCarloResult:
    """نتایج شبیه‌سازی Monte Carlo"""
    simulations: int
    final_equities: List[float]
    percentile_5: float
    percentile_25: float
    percentile_50: float
    percentile_75: float
    percentile_95: float
    mean_return: float
    std_return: float
    prob_profit: float
    prob_ruin: float
    max_drawdowns: List[float]
    worst_case: float
    best_case: float


class MonteCarloSimulator:
    """شبیه‌سازی Monte Carlo"""
    
    def __init__(self, trades: List, initial_capital: float = 10000):
        """
        Args:
            trades: لیست معاملات از backtest
            initial_capital: سرمایه اولیه
        """
        self.trades = trades
        self.initial_capital = initial_capital
        
        # استخراج آمار معاملات
        self.trade_returns = [t.net_profit / initial_capital for t in trades]
        self.trade_pnls = [t.net_profit for t in trades]
        
        print(f"📊 آمار معاملات:")
        print(f"   تعداد معاملات: {len(trades)}")
        print(f"   میانگین سود: ${np.mean(self.trade_pnls):.2f}")
        print(f"   انحراف معیار: ${np.std(self.trade_pnls):.2f}")
    
    def run_simulation(self, n_simulations: int = 1000, 
                      n_trades: int = None,
                      random_seed: int = None) -> MonteCarloResult:
        """
        اجرای شبیه‌سازی Monte Carlo
        
        Args:
            n_simulations: تعداد شبیه‌سازی‌ها
            n_trades: تعداد معاملات در هر شبیه‌سازی (اگر None باشد، از تعداد واقعی استفاده می‌شود)
            random_seed: seed برای تکرارپذیری
        
        Returns:
            MonteCarloResult
        """
        if random_seed:
            np.random.seed(random_seed)
        
        if n_trades is None:
            n_trades = len(self.trades)
        
        print(f"\n🎲 شبیه‌سازی Monte Carlo...")
        print(f"   تعداد شبیه‌سازی: {n_simulations:,}")
        print(f"   تعداد معاملات هر شبیه‌سازی: {n_trades}")
        
        final_equities = []
        max_drawdowns = []
        
        for sim in range(n_simulations):
            if sim % 100 == 0:
                print(f"   پیشرفت: {sim}/{n_simulations}", end='\r')
            
            # انتخاب تصادفی معاملات (با جایگذاری)
            sampled_pnls = np.random.choice(self.trade_pnls, size=n_trades, replace=True)
            
            # محاسبه equity
            equity_curve = [self.initial_capital]
            for pnl in sampled_pnls:
                equity_curve.append(equity_curve[-1] + pnl)
            
            final_equities.append(equity_curve[-1])
            
            # محاسبه Max Drawdown
            equity_array = np.array(equity_curve)
            running_max = np.maximum.accumulate(equity_array)
            drawdown = equity_array - running_max
            max_dd_pct = (np.min(drawdown) / running_max[np.argmin(drawdown)]) * 100
            max_drawdowns.append(max_dd_pct)
        
        print(f"   پیشرفت: {n_simulations}/{n_simulations} ✅")
        
        # محاسبه آمار
        final_equities = np.array(final_equities)
        returns = ((final_equities - self.initial_capital) / self.initial_capital) * 100
        
        result = MonteCarloResult(
            simulations=n_simulations,
            final_equities=final_equities.tolist(),
            percentile_5=np.percentile(final_equities, 5),
            percentile_25=np.percentile(final_equities, 25),
            percentile_50=np.percentile(final_equities, 50),
            percentile_75=np.percentile(final_equities, 75),
            percentile_95=np.percentile(final_equities, 95),
            mean_return=np.mean(returns),
            std_return=np.std(returns),
            prob_profit=(final_equities > self.initial_capital).sum() / n_simulations,
            prob_ruin=(final_equities < self.initial_capital * 0.5).sum() / n_simulations,
            max_drawdowns=max_drawdowns,
            worst_case=np.min(final_equities),
            best_case=np.max(final_equities)
        )
        
        return result

"""
====================================================================
فایل: walk_forward.py
بهینه‌سازی Walk-Forward برای تست Out-of-Sample
====================================================================
"""

import pandas as pd
import numpy as np
from typing import List, Dict, Tuple
from dataclasses import dataclass
import matplotlib.pyplot as plt
from datetime import datetime
import json


@dataclass
class WalkForwardPeriod:
    """یک دوره Walk-Forward"""
    period_number: int
    train_start: datetime
    train_end: datetime
    test_start: datetime
    test_end: datetime
    
    # نتایج آموزش
    train_params: Dict = None
    train_score: float = 0.0
    
    # نتایج تست
    test_trades: int = 0
    test_profit: float = 0.0
    test_win_rate: float = 0.0
    test_profit_factor: float = 0.0
    test_max_dd: float = 0.0
    
    # وضعیت
    is_completed: bool = False
    is_overfitted: bool = False


@dataclass
class WalkForwardResult:
    """نتایج کلی Walk-Forward"""
    periods: List[WalkForwardPeriod]
    total_train_score: float
    total_test_profit: float
    total_test_trades: int
    avg_test_win_rate: float
    avg_test_profit_factor: float
    consistency: float  # سازگاری بین دوره‌ها
    overfitting_score: float  # نمره overfitting


class WalkForwardOptimizer:
    """بهینه‌ساز Walk-Forward"""
    
    def __init__(self, train_func, test_func):
        """
        Args:
            train_func: تابع آموزش که پارامترها را بهینه می‌کند
                        signature: (df_train) -> (best_params, score)
            test_func: تابع تست که با پارامترهای داده شده backtest می‌کند
                       signature: (df_test, params) -> result
        """
        self.train_func = train_func
        self.test_func = test_func
    
    def split_data(self, df: pd.DataFrame, 
                  n_splits: int = 5,
                  train_size: float = 0.6) -> List[WalkForwardPeriod]:
        """
        تقسیم داده به دوره‌های Walk-Forward
        
        Args:
            df: DataFrame با ستون 'time'
            n_splits: تعداد splits
            train_size: نسبت train در هر split
        
        Returns:
            لیست WalkForwardPeriod
        """
        print(f"\n📊 تقسیم داده برای Walk-Forward...")
        print(f"   تعداد splits: {n_splits}")
        print(f"   نسبت train/test: {train_size:.0%}/{1-train_size:.0%}")
        
        total_size = len(df)
        window_size = total_size // n_splits
        
        periods = []
        
        for i in range(n_splits):
            # محاسبه indices
            start_idx = i * window_size
            train_end_idx = start_idx + int(window_size * train_size)
            test_end_idx = start_idx + window_size
            
            # اطمینان از عدم تجاوز
            if test_end_idx > total_size:
                test_end_idx = total_size
            
            # ساخت Period
            period = WalkForwardPeriod(
                period_number=i + 1,
                train_start=df.iloc[start_idx]['time'],
                train_end=df.iloc[train_end_idx-1]['time'],
                test_start=df.iloc[train_end_idx]['time'],
                test_end=df.iloc[test_end_idx-1]['time']
            )
            
            periods.append(period)
            
            print(f"\n   Period {i+1}:")
            print(f"      Train: {period.train_start.date()} → {period.train_end.date()}")
            print(f"      Test:  {period.test_start.date()} → {period.test_end.date()}")
        
        return periods
    
    def run_optimization(self, df: pd.DataFrame,
                        periods: List[WalkForwardPeriod]) -> WalkForwardResult:
        """
        اجرای بهینه‌سازی Walk-Forward
        
        Args:
            df: DataFrame کامل
            periods: لیست دوره‌ها
        
        Returns:
            WalkForwardResult
        """
        print(f"\n{'='*70}")
        print("🔄 شروع بهینه‌سازی Walk-Forward")
        print(f"{'='*70}")
        
        for period in periods:
            print(f"\n{'─'*70}")
            print(f"📊 Period {period.period_number}")
            print(f"{'─'*70}")
            
            # تقسیم داده
            df_train = df[(df['time'] >= period.train_start) & 
                         (df['time'] <= period.train_end)]
            df_test = df[(df['time'] >= period.test_start) & 
                        (df['time'] <= period.test_end)]
            
            print(f"   Train: {len(df_train)} کندل")
            print(f"   Test:  {len(df_test)} کندل")
            
            # آموزش و بهینه‌سازی
            print(f"\n   🔧 آموزش و بهینه‌سازی...")
            best_params, train_score = self.train_func(df_train)
            
            period.train_params = best_params
            period.train_score = train_score
            
            print(f"   ✅ بهترین پارامترها: {best_params}")
            print(f"   📈 نمره آموزش: {train_score:.4f}")
            
            # تست
            print(f"\n   🧪 تست روی Out-of-Sample...")
            test_result = self.test_func(df_test, best_params)
            
            period.test_trades = test_result['trades']
            period.test_profit = test_result['profit']
            period.test_win_rate = test_result['win_rate']
            period.test_profit_factor = test_result['profit_factor']
            period.test_max_dd = test_result['max_dd']
            period.is_completed = True
            
            print(f"   📊 نتایج تست:")
            print(f"      معاملات: {period.test_trades}")
            print(f"      سود: ${period.test_profit:,.2f}")
            print(f"      Win Rate: {period.test_win_rate*100:.1f}%")
            print(f"      Profit Factor: {period.test_profit_factor:.2f}")
            
            # تشخیص Overfitting
            if train_score > 0 and test_result['profit'] < 0:
                period.is_overfitted = True
                print(f"   ⚠️ احتمال Overfitting!")
        
        # تجمیع نتایج
        result = self._aggregate_results(periods)
        
        return result
    
    def _aggregate_results(self, periods: List[WalkForwardPeriod]) -> WalkForwardResult:
        """تجمیع نتایج تمام دوره‌ها"""
        
        # جمع آوری آمار
        train_scores = [p.train_score for p in periods if p.is_completed]
        test_profits = [p.test_profit for p in periods if p.is_completed]
        test_win_rates = [p.test_win_rate for p in periods if p.is_completed]
        test_profit_factors = [p.test_profit_factor for p in periods if p.is_completed]
        
        # محاسبه consistency (انحراف معیار سود تست)
        consistency = 1 - (np.std(test_profits) / (np.abs(np.mean(test_profits)) + 1))
        consistency = max(0, min(1, consistency))
        
        # محاسبه overfitting score
        # اگر نمره train بالا ولی profit test پایین → overfitting
        overfit_cases = sum([1 for p in periods if p.is_overfitted])
        overfitting_score = overfit_cases / len(periods)
        
        result = WalkForwardResult(
            periods=periods,
            total_train_score=np.mean(train_scores),
            total_test_profit=sum(test_profits),
            total_test_trades=sum([p.test_trades for p in periods]),
            avg_test_win_rate=np.mean(test_win_rates),
            avg_test_profit_factor=np.mean(test_profit_factors),
            consistency=consistency,
            overfitting_score=overfitting_score
        )
        
        return result
    
    def plot_results(self, result: WalkForwardResult, save_path: str = None):
        """رسم نتایج Walk-Forward"""
        fig, axes = plt.subplots(2, 2, figsize=(15, 10))
        
        periods = result.periods
        period_nums = [p.period_number for p in periods]
        
        # 1. نمره Train vs Test Profit
        ax1 = axes[0, 0]
        train_scores = [p.train_score for p in periods]
        test_profits = [p.test_profit for p in periods]
        
        ax1_twin = ax1.twinx()
        
        bars1 = ax1.bar([x - 0.2 for x in period_nums], train_scores, 
                       width=0.4, label='Train Score', 
                       color='blue', alpha=0.7)
        bars2 = ax1_twin.bar([x + 0.2 for x in period_nums], test_profits,
                            width=0.4, label='Test Profit',
                            color='green' if sum(test_profits) > 0 else 'red',

        ax1.set_xlabel('Period', fontweight='bold')
        ax1.set_ylabel('Train Score', fontweight='bold', color='blue')
        ax1_twin.set_ylabel('Test Profit ($)', fontweight='bold', color='green')

        ax1_twin.tick_params(axis='y', labelcolor='green')
        ax1.grid(True, alpha=0.3)
        
        # 2. Win Rate در هر دوره
        ax2 = axes[0, 1]
        win_rates = [p.test_win_rate * 100 for p in periods]
        colors = ['green' if wr > 50 else 'red' for wr in win_rates]
        
        ax2.bar(period_nums, win_rates, color=colors, alpha=0.7, edgecolor='black')
        ax2.axhline(y=50, color='black', linestyle='--', linewidth=1)
        ax2.set_xlabel('Period', fontweight='bold')
        ax2.set_ylabel('Win Rate (%)', fontweight='bold')
        ax2.set_title('Win Rate per Period', fontweight='bold')
        ax2.grid(True, alpha=0.3, axis='y')
        
        # 3. Profit Factor
        ax3 = axes[1, 0]
        pfs = [p.test_profit_factor for p in periods]
        colors = ['green' if pf > 1.5 else 'orange' if pf > 1 else 'red' for pf in pfs]
        
        ax3.bar(period_nums, pfs, color=colors, alpha=0.7, edgecolor='black')
        ax3.axhline(y=1.0, color='black', linestyle='--', linewidth=1, label='Break-even')
        ax3.axhline(y=1.5, color='green', linestyle='--', linewidth=1, label='Good')
        ax3.set_xlabel('Period', fontweight='bold')
        ax3.set_ylabel('Profit Factor', fontweight='bold')
        ax3.set_title('Profit Factor per Period', fontweight='bold')
        ax3.legend()
        ax3.grid(True, alpha=0.3, axis='y')
        
        # 4. خلاصه (متنی)
        ax4 = axes[1, 1]
        ax4.axis('off')
        
        summary_text = f"""
WALK-FORWARD OPTIMIZATION RESULTS
{'='*40}

Periods: {len(periods)}

TRAIN
Avg Score: {result.total_train_score:.4f}

TEST (Out-of-Sample)
Total Profit:  ${result.total_test_profit:>10,.2f}
Total Trades:  {result.total_test_trades:>10}
Avg Win Rate:  {result.avg_test_win_rate*100:>9.1f}%
Avg PF:        {result.avg_test_profit_factor:>10.2f}

ROBUSTNESS
Consistency:   {result.consistency*100:>9.1f}%
Overfitting:   {result.overfitting_score*100:>9.1f}%

EVALUATION
"""
        
        # ارزیابی
        if result.overfitting_score < 0.2 and result.consistency > 0.6:
            eval_text = "ROBUST ✅"
            eval_color = 'green'
        elif result.overfitting_score < 0.4:
            eval_text = "ACCEPTABLE ⚠️"
            eval_color = 'orange'
        else:
            eval_text = "OVERFITTED ❌"
            eval_color = 'red'
        
        summary_text += f"{eval_text}"
        
        ax4.text(0.05, 0.95, summary_text,
                fontsize=10, fontfamily='monospace',
                verticalalignment='top',
                transform=ax4.transAxes)
        
        # رنگ ارزیابی
        ax4.text(0.5, 0.1, eval_text,
                fontsize=16, fontweight='bold',
                color=eval_color,
                ha='center',
                transform=ax4.transAxes)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"✅ نمودار Walk-Forward ذخیره شد: {save_path}")
        
        return fig
