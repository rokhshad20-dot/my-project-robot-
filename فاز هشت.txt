"""

فایل: dynamic_position_manager.py
مدیریت پوزیشن با SL/TP پویا و Trailing Stop هوشمند
====================================================================
"""

import MetaTrader5 as mt5
from dataclasses import dataclass
from datetime import datetime
from typing import Dict, Optional, List
import numpy as np


@dataclass
class PositionState:
    """وضعیت یک پوزیشن"""
    ticket: int
    symbol: str
    direction: str  # 'BUY' or 'SELL'
    volume: float
    entry_price: float
    current_sl: float
    current_tp: float
    
    # متریک‌ها
    open_time: datetime
    highest_profit: float = 0.0
    lowest_profit: float = 0.0
    max_favorable_pips: float = 0.0
    max_adverse_pips: float = 0.0
    
    # وضعیت مدیریت
    be_moved: bool = False
    partial_closed: bool = False
    partial_volume: float = 0.0
    trailing_active: bool = False
    
    # تنظیمات
    atr: float = 0.0
    risk_reward: float = 2.0


class DynamicPositionManager:
    """
    مدیریت پوزیشن با قابلیت‌های پیشرفته:
    
    1. Break Even خودکار
    2. Partial Close هوشمند
    3. Trailing Stop چند مرحله‌ای
    4. SL/TP پویا بر اساس نوسان
    """
    
    def __init__(self, config: Dict = None):
        """
        Args:
            config: تنظیمات شامل:
                - be_trigger: فاصله برای فعال شدن BE (برحسب ATR)
                - partial_trigger: فاصله برای Partial Close
                - partial_percent: درصد بستن در Partial
                - trailing_levels: سطوح مختلف Trailing
        """
        self.config = config or {}
        
        # تنظیمات پیش‌فرض
        self.be_trigger = self.config.get('be_trigger', 1.5)  # 1.5x ATR
        self.partial_trigger = self.config.get('partial_trigger', 2.0)  # 2x ATR
        self.partial_percent = self.config.get('partial_percent', 0.5)  # 50%
        
        # سطوح Trailing Stop
        self.trailing_levels = self.config.get('trailing_levels', [
            {'profit_atr': 1.5, 'trail_atr': 0.5},  # سود 1.5 ATR → Trail 0.5 ATR
            {'profit_atr': 2.5, 'trail_atr': 1.0},  # سود 2.5 ATR → Trail 1.0 ATR
            {'profit_atr': 4.0, 'trail_atr': 1.5},  # سود 4.0 ATR → Trail 1.5 ATR
        ])
        
        # پوزیشن‌های تحت مدیریت
        self.managed_positions: Dict[int, PositionState] = {}
        
        print("✅ Dynamic Position Manager راه‌اندازی شد")
        print(f"   BE Trigger: {self.be_trigger}x ATR")
        print(f"   Partial Trigger: {self.partial_trigger}x ATR")
        print(f"   Partial %: {self.partial_percent*100:.0f}%")
    
    def add_position(self, ticket: int, symbol: str, direction: str,
                    volume: float, entry_price: float, sl: float, 
                    tp: float, atr: float):
        """
        اضافه کردن پوزیشن جدید برای مدیریت
        
        Args:
            ticket: شماره تیکت
            symbol: نماد
            direction: جهت
            volume: حجم
            entry_price: قیمت ورود
            sl: Stop Loss
            tp: Take Profit
            atr: ATR فعلی
        """
        position = PositionState(
            ticket=ticket,
            symbol=symbol,
            direction=direction,
            volume=volume,
            entry_price=entry_price,
            current_sl=sl,
            current_tp=tp,
            open_time=datetime.now(),
            atr=atr,
            risk_reward=abs(tp - entry_price) / abs(entry_price - sl)
        )
        
        self.managed_positions[ticket] = position
        
        print(f"📌 پوزیشن #{ticket} اضافه شد به مدیریت")
        print(f"   {direction} {volume} @ {entry_price:.5f}")
        print(f"   SL: {sl:.5f} | TP: {tp:.5f}")
    
    def update_all_positions(self, current_data: Dict):
        """
        به‌روزرسانی تمام پوزیشن‌های باز
        
        Args:
            current_data: دیکشنری شامل:
                - price: قیمت فعلی
                - atr: ATR فعلی
                - adx: ADX فعلی (اختیاری)
        """
        if not mt5.initialize():
            print("❌ خطا: نمی‌توان به MT5 متصل شد")
            return
        
        # دریافت پوزیشن‌های باز
        positions = mt5.positions_get()
        
        if not positions:
            mt5.shutdown()
            return
        
        for pos in positions:
            if pos.ticket in self.managed_positions:
                # به‌روزرسانی این پوزیشن
                self._update_position(pos, current_data)
        
        mt5.shutdown()
    
    def _update_position(self, mt5_pos, current_data: Dict):
        """به‌روزرسانی یک پوزیشن"""
        
        position = self.managed_positions[mt5_pos.ticket]
        
        # قیمت فعلی
        current_price = current_data['price']
        
        # محاسبه سود فعلی (به پیپ)
        if position.direction == 'BUY':
            profit_pips = (current_price - position.entry_price) * 10000
        else:  # SELL
            profit_pips = (position.entry_price - current_price) * 10000
        
        # به‌روزرسانی متریک‌ها
        position.highest_profit = max(position.highest_profit, mt5_pos.profit)
        position.lowest_profit = min(position.lowest_profit, mt5_pos.profit)
        position.max_favorable_pips = max(position.max_favorable_pips, profit_pips)
        
        if profit_pips < 0:
            position.max_adverse_pips = min(position.max_adverse_pips, profit_pips)
        
        # ATR فعلی
        current_atr = current_data.get('atr', position.atr)
        atr_pips = current_atr * 10000
        
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        # 1️⃣ بررسی Break Even
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        
        if not position.be_moved and profit_pips > self.be_trigger * atr_pips:
            self._move_to_breakeven(position, current_price)
        
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        # 2️⃣ بررسی Partial Close
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        
        if not position.partial_closed and profit_pips > self.partial_trigger * atr_pips:
            self._partial_close(position, mt5_pos)
        
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        # 3️⃣ بررسی Trailing Stop
        # ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        
        if position.be_moved:
            self._update_trailing_stop(position, current_price, current_atr)
    
    def _move_to_breakeven(self, position: PositionState, current_price: float):
        """انتقال SL به نقطه سر به سر"""
        
        # محاسبه BE با یک buffer کوچک
        buffer_pips = 2  # 2 پیپ بالاتر از entry
        buffer = buffer_pips * 0.0001
        
        if position.direction == 'BUY':
            new_sl = position.entry_price + buffer
        else:  # SELL
            new_sl = position.entry_price - buffer
        
        # بررسی که SL جدید بهتر از SL فعلی باشد
        if position.direction == 'BUY':
            if new_sl <= position.current_sl:
                return
        else:
            if new_sl >= position.current_sl:
                return
        
        # تغییر SL
        success = self._modify_position(position.ticket, new_sl, position.current_tp)
        
        if success:
            position.current_sl = new_sl
            position.be_moved = True
            
            print(f"\n📌 Break Even: #{position.ticket}")
            print(f"   SL منتقل شد به: {new_sl:.5f}")
            print(f"   سود فعلی: {position.highest_profit:.2f}$")
    
    def _partial_close(self, position: PositionState, mt5_pos):
        """بستن جزئی پوزیشن"""
        
        # حجم برای بستن
        close_volume = position.volume * self.partial_percent
        
        # گرد کردن به حداقل حجم
        close_volume = round(close_volume, 2)
        
        if close_volume < 0.01:
            return
        
        # بستن
        success = self._close_partial(
            position.ticket,
            close_volume,
            position.direction
        )
        
        if success:
            position.partial_closed = True
            position.partial_volume = close_volume
            position.volume -= close_volume
            
            print(f"\n💰 Partial Close: #{position.ticket}")
            print(f"   {close_volume} لات ({self.partial_percent*100:.0f}%) بسته شد")
            print(f"   سود قفل شده: ~${position.highest_profit * self.partial_percent:.2f}")
            print(f"   باقی‌مانده: {position.volume} لات")
    
    def _update_trailing_stop(self, position: PositionState, 
                              current_price: float, current_atr: float):
        """
        به‌روزرسانی Trailing Stop چند مرحله‌ای
        """
        # محاسبه سود به ATR
        if position.direction == 'BUY':
            profit_atr = (current_price - position.entry_price) / current_atr
        else:
            profit_atr = (position.entry_price - current_price) / current_atr
        
        if profit_atr <= 0:
            return
        
        # پیدا کردن سطح مناسب
        trail_distance = None
        
        for level in sorted(self.trailing_levels, key=lambda x: x['profit_atr'], reverse=True):
            if profit_atr >= level['profit_atr']:
                trail_distance = level['trail_atr'] * current_atr
                break
        
        if trail_distance is None:
            return
        
        # محاسبه SL جدید
        if position.direction == 'BUY':
            new_sl = current_price - trail_distance
        else:
            new_sl = current_price + trail_distance
        
        # بررسی که SL جدید بهتر باشد
        if position.direction == 'BUY':
            if new_sl <= position.current_sl:
                return
        else:
            if new_sl >= position.current_sl:
                return
        
        # تغییر SL
        success = self._modify_position(position.ticket, new_sl, position.current_tp)
        
        if success:
            old_sl = position.current_sl
            position.current_sl = new_sl
            position.trailing_active = True
            
            sl_move_pips = abs(new_sl - old_sl) * 10000
            
            print(f"\n🎯 Trailing Stop: #{position.ticket}")
            print(f"   SL: {old_sl:.5f} → {new_sl:.5f} (+{sl_move_pips:.1f} پیپ)")
            print(f"   فاصله از قیمت: {trail_distance*10000:.1f} پیپ")
    
    def _modify_position(self, ticket: int, new_sl: float, new_tp: float) -> bool:
        """تغییر SL/TP یک پوزیشن"""
        
        if not mt5.initialize():
            return False
        
        request = {
            "action": mt5.TRADE_ACTION_SLTP,
            "position": ticket,
            "sl": new_sl,
            "tp": new_tp,
        }
        
        result = mt5.order_send(request)
        mt5.shutdown()
        
        if result.retcode == mt5.TRADE_RETCODE_DONE:
            return True
        else:
            print(f"⚠️ خطا در تغییر SL/TP: {result.comment}")
            return False
    
    def _close_partial(self, ticket: int, volume: float, direction: str) -> bool:
        """بستن جزئی یک پوزیشن"""
        
        if not mt5.initialize():
            return False
        
        # نوع معکوس برای بستن
        if direction == 'BUY':
            order_type = mt5.ORDER_TYPE_SELL
        else:
            order_type = mt5.ORDER_TYPE_BUY
        
        # دریافت قیمت فعلی
        position = mt5.positions_get(ticket=ticket)
        
        if not position:
            mt5.shutdown()
            return False
        
        pos = position[0]
        
        request = {
            "action": mt5.TRADE_ACTION_DEAL,
            "position": ticket,
            "symbol": pos.symbol,
            "volume": volume,
            "type": order_type,
            "deviation": 20,
            "magic": pos.magic,
            "comment": "Partial_Close",
            "type_filling": mt5.ORDER_FILLING_IOC,
        }
        
        result = mt5.order_send(request)
        mt5.shutdown()
        
        if result.retcode == mt5.TRADE_RETCODE_DONE:
            return True
        else:
            print(f"⚠️ خطا در Partial Close: {result.comment}")
            return False
    
    def get_position_summary(self, ticket: int) -> Dict:
        """دریافت خلاصه یک پوزیشن"""
        
        if ticket not in self.managed_positions:
            return {}
        
        position = self.managed_positions[ticket]
        
        return {
            'ticket': position.ticket,
            'direction': position.direction,
            'volume': position.volume,
            'entry_price': position.entry_price,
            'current_sl': position.current_sl,
            'current_tp': position.current_tp,
            'be_moved': position.be_moved,
            'partial_closed': position.partial_closed,
            'partial_volume': position.partial_volume,
            'trailing_active': position.trailing_active,
            'highest_profit': position.highest_profit,
            'max_favorable_pips': position.max_favorable_pips,
            'max_adverse_pips': position.max_adverse_pips,
            'duration': (datetime.now() - position.open_time).seconds // 60
        }
    
    def print_all_positions(self):
        """چاپ وضعیت تمام پوزیشن‌ها"""
        
        if not self.managed_positions:
            print("📭 هیچ پوزیشن فعالی تحت مدیریت نیست")
            return
        
        print("\n" + "="*70)
        print("📊 پوزیشن‌های تحت مدیریت")
        print("="*70)
        
        for ticket, position in self.managed_positions.items():
            print(f"\n🎫 #{ticket} - {position.direction} {position.volume}")
            print(f"   ورود: {position.entry_price:.5f}")
            print(f"   SL: {position.current_sl:.5f} | TP: {position.current_tp:.5f}")
            print(f"   BE: {'✅' if position.be_moved else '❌'} | "
                  f"Partial: {'✅' if position.partial_closed else '❌'} | "
                  f"Trailing: {'✅' if position.trailing_active else '❌'}")
            print(f"   سود بیشینه: ${position.highest_profit:.2f}")
            print(f"   MFE: {position.max_favorable_pips:.1f} پیپ | "
                  f"MAE: {position.max_adverse_pips:.1f} پیپ")
        
        print("="*70)
    
    def remove_closed_positions(self):
        """حذف پوزیشن‌های بسته شده از لیست"""
        
        if not mt5.initialize():
            return
        
        open_tickets = {pos.ticket for pos in mt5.positions_get() or []}
        mt5.shutdown()
        
        # پیدا کردن بسته‌شده‌ها
        closed = [ticket for ticket in self.managed_positions.keys() 
                 if ticket not in open_tickets]
        
        for ticket in closed:
            del self.managed_positions[ticket]
            print(f"🏁 پوزیشن #{ticket} بسته شد - حذف از مدیریت")

            print("   ✅ مدل بارگذاری شد")
            
            # بررسی نیاز به Retrain
            should_retrain, reason = self.model_monitor.should_retrain()
            if should_retrain:
                self.logger.log_warning(
                    'MODEL_RETRAIN_NEEDED',
                    f'مدل نیاز به Retrain دارد: {reason}'
                )
                self.alert_system.send_alert(Alert(
                    alert_type='RETRAIN_NEEDED',
                    severity='WARNING',
                    title='نیاز به آموزش مجدد',
                    message=reason,
                    data={'recommendation': 'لطفاً train.py را اجرا کنید'}
                ))
        
        except Exception as e:
            self.logger.log_error(
                'MODEL_LOAD',
                f'خطا در بارگذاری مدل: {e}',
                'ابتدا train.py را اجرا کنید'
            )
            self.alert_system.alert_system_error('MODEL_LOAD', str(e))
            return False
        
        # 2. تست اتصال MT5
        print("\n2️⃣ بررسی اتصال MT5...")
        if self.mode == 'live':
            if not mt5.initialize():
                self.logger.log_error(
                    'MT5_CONNECTION',
                    'نمی‌توان به MT5 متصل شد',
                    'MT5 را به عنوان Administrator اجرا کنید'

            mt5.shutdown()

        # 3. بازیابی Session قبلی
        print("\n3️⃣ بررسی Session قبلی...")
        self._recover_session()
        
        # 4. ثبت شروع
        self.session_start = datetime.now()
        self.logger.log_info(
            'BOT_STARTED',
            f'ربات در حالت {self.mode} شروع شد'
        )
        
        return True
    
    def get_signal_with_analysis(self) -> Optional[Dict]:
        """
        دریافت سیگنال با تحلیل کامل
        
        جریان کار:
        1. دریافت داده
        2. پیش‌بینی ML
        3. تحلیل ترکیبی (Hybrid)
        4. بررسی فیلترها
        5. محاسبه SL/TP پویا
        
        Returns:
            Dict یا None
        """
        try:
            # گام 1: دریافت داده
            df = prepare_live_data(Config.SYMBOL, Config.TIMEFRAME_PRIMARY, 200)
            
            if df is None or len(df) < Config.LOOKBACK_PERIOD:
                self.logger.log_error(
                    'DATA_ERROR',
                    'داده کافی دریافت نشد',
                    'بررسی کنید MT5 و اتصال اینترنت'
                )
                return None
            
            # گام 2: پیش‌بینی ML
            features = df[self.model.features].iloc[-Config.LOOKBACK_PERIOD:].values
            X_scaled = self.model.scaler.transform(features)
            X = X_scaled.reshape(1, Config.LOOKBACK_PERIOD, -1)
            prediction = self.model.predict(X)
            
            # محاسبه confidence
            ml_confidence = abs(prediction - 0.5) * 2
            
            # تعیین جهت اولیه
            if prediction > Config.BUY_THRESHOLD:
                ml_direction = 'BUY'
            elif prediction < Config.SELL_THRESHOLD:
                ml_direction = 'SELL'
            else:
                ml_direction = 'NONE'
            
            # گام 3: لاگ وضعیت بازار
            last_row = df.iloc[-1]
            self.logger.log_market_snapshot(
                price=last_row['close'],
                atr=last_row['ATR'],
                adx=last_row['ADX'],
                rsi=last_row['RSI'],
                market_state=self._determine_market_state(last_row)
            )
            
            # گام 4: لاگ نظر ML
            self.logger.log_ml_decision(
                prediction=prediction,
                confidence=ml_confidence,
                direction=ml_direction,
                reasoning=self._ml_reasoning(prediction, last_row)
            )
            
            # ثبت در Model Monitor
            self.model_monitor.log_prediction(
                prediction=prediction,
                confidence=ml_confidence,
                direction=ml_direction,
                notes=f"Cycle #{self.cycle_count}"
            )
            
            # گام 5: تحلیل ترکیبی (Hybrid)
            print("\n🔄 تحلیل ترکیبی ML + Technical...")
            
            hybrid_decision = self.hybrid_analyzer.analyze(
                df=df,
                ml_prediction=prediction,
                ml_confidence=ml_confidence
            )
            
            # لاگ تصمیم Hybrid
            if not hybrid_decision.should_trade:
                reasons = hybrid_decision.reasons + hybrid_decision.warnings
                self.logger.log_why_not_trade(
                    reasons=reasons,
                    details={
                        'prediction': float(prediction),
                        'ml_direction': ml_direction,
                        'confidence': float(ml_confidence),
                        'patterns': hybrid_decision.patterns_found,
                        'near_support': hybrid_decision.near_support,
                        'near_resistance': hybrid_decision.near_resistance
                    }
                )
                return None
            
            # گام 6: بررسی ریسک
            can_trade, risk_reason = self.risk.can_trade()
            
            if not can_trade:
                self.logger.log_why_not_trade(
                    reasons=[f"مدیریت ریسک: {risk_reason}"],
                    details={'risk_check': risk_reason}
                )
                return None
            
            # گام 7: دریافت قیمت واقعی
            if self.mode == 'live':
                if not mt5.initialize():
                    self.logger.log_error('MT5_CONNECTION', 'قطع اتصال MT5', None)
                    return None
                
                tick = mt5.symbol_info_tick(Config.SYMBOL)
                mt5.shutdown()
                
                if not tick:
                    return None
                
                if hybrid_decision.direction == 'BUY':
                    entry = tick.ask
                else:
                    entry = tick.bid
            else:
                # حالت Backtest
                entry = last_row['close']
            
            # گام 8: محاسبه SL/TP پویا
            current_atr = last_row['ATR']
            current_adx = last_row['ADX']
            avg_atr = df['ATR'].iloc[-20:].mean()
            
            sl, tp, sl_dist, tp_dist = self.risk.calculate_dynamic_sl_tp(
                entry_price=entry,
                direction=hybrid_decision.direction,
                current_atr=current_atr,
                avg_atr=avg_atr,
                adx=current_adx
            )
            
            # گام 9: محاسبه حجم
            volume = self.risk.calculate_volume(entry, sl)
            
            # گام 10: ساخت سیگنال نهایی
            signal = {
                'direction': hybrid_decision.direction,
                'entry': entry,
                'sl': sl,
                'tp': tp,
                'volume': volume,
                'atr': current_atr,
                'adx': current_adx,
                
                # اطلاعات تکمیلی
                'ml_prediction': prediction,
                'ml_confidence': ml_confidence,
                'hybrid_confidence': hybrid_decision.confidence,
                'patterns': hybrid_decision.patterns_found,
                'reasons': hybrid_decision.reasons,
                'warnings': hybrid_decision.warnings,
                
                # متریک‌ها
                'sl_pips': sl_dist * 10000,
                'tp_pips': tp_dist * 10000,
                'risk_reward': tp_dist / sl_dist if sl_dist > 0 else 0
            }
            
            return signal
        
        except Exception as e:
            self.logger.log_error(
                'SIGNAL_ERROR',
                f'خطا در تولید سیگنال: {e}',
                'بررسی کنید همه فایل‌ها موجود باشند'
            )
            traceback.print_exc()
            return None
    
    def execute_trade(self, signal: Dict) -> bool:
        """اجرای معامله با لاگ کامل"""
        
        try:
            if self.mode == 'backtest':
                return self._simulate_trade(signal)
            
            # معامله واقعی MT5
            if not mt5.initialize():
                self.logger.log_error('MT5_CONNECTION', 'قطع اتصال MT5', None)
                self.alert_system.alert_connection_lost('MT5')
                return False
            
            order_type = mt5.ORDER_TYPE_BUY if signal['direction'] == 'BUY' else mt5.ORDER_TYPE_SELL
            
            request = {
                "action": mt5.TRADE_ACTION_DEAL,
                "symbol": Config.SYMBOL,
                "volume": signal['volume'],
                "type": order_type,
                "price": signal['entry'],
                "sl": signal['sl'],
                "tp": signal['tp'],
                "deviation": 20,
                "magic": 123456,
                "comment": "Advanced_Bot_v2",
                "type_filling": mt5.ORDER_FILLING_IOC,
            }
            
            result = mt5.order_send(request)
            mt5.shutdown()
            
            if result.retcode == mt5.TRADE_RETCODE_DONE:
                # هشدار موفقیت
                self.alert_system.alert_trade_opened(
                    signal['direction'],
                    signal['entry'],
                    signal['volume']
                )
                
                # لاگ جامع
                self.logger.log_trade_open(
                    direction=signal['direction'],
                    price=signal['entry'],
                    volume=signal['volume'],
                    sl=signal['sl'],
                    tp=signal['tp'],
                    reason=', '.join(signal.get('reasons', ['ML Signal'])),
                    risk_reward=signal['risk_reward']
                )
                
                # اضافه به Position Manager
                self.position_manager.add_position(
                    ticket=result.order,
                    symbol=Config.SYMBOL,
                    direction=signal['direction'],
                    volume=signal['volume'],
                    entry_price=signal['entry'],
                    sl=signal['sl'],
                    tp=signal['tp'],
                    atr=signal['atr']
                )
                
                # آمار
                self.daily_trades += 1
                self._save_session()
                
                return True
            
            else:
                # خطا در اجرا
                error_msg = f"خطا در باز کردن معامله: {result.comment}"
                self.logger.log_error('TRADE_EXECUTION', error_msg, None)
                self.alert_system.alert_system_error('TRADE_EXECUTION', error_msg)
                return False
        
        except Exception as e:
            self.logger.log_error('TRADE_EXECUTION', str(e), 'بررسی MT5')
            traceback.print_exc()
            return False
    
    def manage_positions(self):
        """مدیریت هوشمند پوزیشن‌های باز"""
        
        try:
            # دریافت داده فعلی
            df = prepare_live_data(Config.SYMBOL, Config.TIMEFRAME_PRIMARY, 50)
            
            if df is None or len(df) == 0:
                return
            
            last_row = df.iloc[-1]
            
            current_data = {
                'price': last_row['close'],
                'atr': last_row['ATR'],
                'adx': last_row.get('ADX', 0)
            }
            
            # به‌روزرسانی با لاگ
            print("   📊 به‌روزرسانی پوزیشن‌ها...")
            self.position_manager.update_all_positions(current_data)
            
            # حذف بسته‌شده‌ها
            self.position_manager.remove_closed_positions()
        
        except Exception as e:
            self.logger.log_error('POSITION_MANAGEMENT', str(e), None)
    
    def check_daily_reset(self):
        """بررسی و ریست آمار روزانه"""
        today = datetime.now().date()
        
        if today > self.last_daily_reset:
            # محاسبه متریک‌های روزانه
            self.model_monitor.calculate_daily_metrics()
            
            # ریست آمار
            self.daily_trades = 0
            self.last_daily_reset = today
            
            # لاگ
            self.logger.log_info(
                'DAILY_RESET',
                f'آمار روزانه ریست شد - تاریخ: {today}'
            )
    
    def run_cycle(self):
        """یک چرخه کامل با لاگینگ جامع"""
        
        self.cycle_count += 1
        
        print("\n" + "━"*70)
        print(f"🔄 چرخه #{self.cycle_count} - {datetime.now().strftime('%H:%M:%S')}")
        print("━"*70)
        
        try:
            # 1. بررسی ریست روزانه
            self.check_daily_reset()
            
            # 2. مدیریت پوزیشن‌های موجود
            print("📊 مدیریت پوزیشن‌های باز...")
            self.manage_positions()
            
            # نمایش وضعیت
            if len(self.position_manager.managed_positions) > 0:
                self.position_manager.print_all_positions()
            else:
                print("   📭 هیچ پوزیشن بازی وجود ندارد")
            
            # 3. بررسی سیگنال جدید
            print("\n🔍 بررسی سیگنال جدید...")
            
            # بررسی محدودیت تعداد پوزیشن
            if len(self.position_manager.managed_positions) >= Config.MAX_POSITIONS:
                print(f"   ⚠️  حداکثر {Config.MAX_POSITIONS} پوزیشن باز است")
                return
            
            # دریافت سیگنال
            signal = self.get_signal_with_analysis()
            
            if signal:
                print(f"\n✅ سیگنال {signal['direction']} یافت شد!")
                print(f"   اطمینان ML: {signal['ml_confidence']*100:.1f}%")
                print(f"   اطمینان Hybrid: {signal['hybrid_confidence']*100:.1f}%")
                print(f"   SL: {signal['sl_pips']:.1f} پیپ")
                print(f"   TP: {signal['tp_pips']:.1f} پیپ")
                print(f"   R:R: 1:{signal['risk_reward']:.2f}")
                
                # اجرای معامله
                self.execute_trade(signal)
            else:
                print("   ⏸️  بدون سیگنال - منتظر فرصت بهتر")
        
        except Exception as e:
            self.logger.log_error('CYCLE_ERROR', str(e), 'خطا در چرخه اصلی')
            traceback.print_exc()

    def _determine_market_state(self, row) -> str:
        """تعیین وضعیت کلی بازار"""
        adx = row['ADX']
        rsi = row['RSI']
        
        if adx > 40:
            trend = "روند قوی"
        elif adx > 25:
            trend = "روند متوسط"
        else:
            trend = "رنج"
        
        if rsi > 60:
            direction = "صعودی"
        elif rsi < 40:
            direction = "نزولی"
        else:
            direction = "خنثی"
        
        return f"{trend} - {direction}"
    
    def _ml_reasoning(self, prediction: float, row) -> str:
        """دلیل تصمیم ML"""
        if prediction > 0.75:
            return f"الگوی صعودی قوی + ADX={row['ADX']:.1f}"
        elif prediction > 0.65:
            return f"الگوی صعودی + شرایط مناسب"
        elif prediction < 0.25:
            return f"الگوی نزولی قوی + ADX={row['ADX']:.1f}"
        elif prediction < 0.35:
            return f"الگوی نزولی + شرایط مناسب"
        else:
            return "عدم اطمینان - بازار مبهم"
    
    def _simulate_trade(self, signal: Dict) -> bool:
        """شبیه‌سازی معامله برای Backtest"""
        print(f"\n🧪 [BACKTEST] شبیه‌سازی معامله {signal['direction']}")
        
        # فرض: تیکت تصادفی
        ticket = int(time.time() * 1000) % 1000000
        
        # اضافه به Position Manager
        self.position_manager.add_position(
            ticket=ticket,
            symbol=Config.SYMBOL,
            direction=signal['direction'],
            volume=signal['volume'],
            entry_price=signal['entry'],
            sl=signal['sl'],
            tp=signal['tp'],
            atr=signal['atr']
        )
        
        return True
    
    def _save_session(self):
        """ذخیره وضعیت Session"""
        try:
            session_data = {
                'session_start': self.session_start.isoformat() if self.session_start else None,
                'cycle_count': self.cycle_count,
                'daily_trades': self.daily_trades,
                'last_daily_reset': self.last_daily_reset.isoformat(),
                'managed_positions': len(self.position_manager.managed_positions)
            }
            
            with open(self.session_file, 'w') as f:
                json.dump(session_data, f, indent=2)
        
        except Exception as e:
            print(f"⚠️  خطا در ذخیره Session: {e}")
    
    def _recover_session(self):
        """بازیابی Session قبلی"""
        try:
            if self.session_file.exists():
                with open(self.session_file, 'r') as f:
                    data = json.load(f)
                
                self.cycle_count = data.get('cycle_count', 0)
                self.daily_trades = data.get('daily_trades', 0)
                
                print(f"   ✅ Session قبلی بازیابی شد")
                print(f"   چرخه قبلی: {self.cycle_count}")
                print(f"   معاملات امروز: {self.daily_trades}")
        
        except Exception as e:
            print(f"   ⚠️  خطا در بازیابی Session: {e}")
    
    def _cleanup(self):
        """پاکسازی و خلاصه نهایی"""
        print("\n" + "="*70)
        print("📊 خلاصه نهایی عملکرد")
        print("="*70)
        
        # مدت زمان
        if self.session_start:
            duration = datetime.now() - self.session_start
            hours = duration.seconds // 3600
            minutes = (duration.seconds % 3600) // 60
            print(f"\n⏱️  مدت اجرا: {hours}h {minutes}m")
        
        print(f"🔄 تعداد چرخه‌ها: {self.cycle_count}")
        print(f"💼 معاملات امروز: {self.daily_trades}")
        
        # وضعیت پوزیشن‌ها
        if len(self.position_manager.managed_positions) > 0:
            print(f"\n📌 پوزیشن‌های باز: {len(self.position_manager.managed_positions)}")
            for pos in self.position_manager.managed_positions.values():
                print(f"   • #{pos.ticket}: {pos.direction} {pos.volume:.2f} لات")
        else:
            print(f"\n📭 هیچ پوزیشن بازی نیست")
        
        # متریک‌های مدل
        print("\n🧠 عملکرد مدل ML:")
        daily_metrics = self.model_monitor.get_daily_metrics()
        if daily_metrics:
            print(f"   Accuracy امروز: {daily_metrics.get('accuracy', 0)*100:.1f}%")
            print(f"   پیش‌بینی‌های انجام شده: {daily_metrics.get('total_predictions', 0)}")
        
        # ثبت پایانی
        self.logger.log_info(
            'BOT_SHUTDOWN',
            f'ربات متوقف شد - {self.cycle_count} چرخه، {self.daily_trades} معامله'
        )
        
        print("\n" + "="*70)
        print("👋 ربات با موفقیت متوقف شد")
        print("="*70)
