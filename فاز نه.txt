"""
====================================================================
فایل: run_tests.py
اجرای کامل تمام تست‌ها با گزارش Coverage
====================================================================
"""

import subprocess
import sys
from pathlib import Path


def run_tests():
    """اجرای تمام تست‌ها"""
    
    print("\n" + "="*70)
    print("🧪 شروع اجرای تست‌های کامل")
    print("="*70)
    
    # مسیر tests
    tests_dir = Path(__file__).parent / "tests"
    
    if not tests_dir.exists():
        print("❌ پوشه tests یافت نشد!")
        return False
    
    # اجرای pytest با coverage
    cmd = [
        sys.executable,
        "-m",
        "pytest",
        str(tests_dir),
        "-v",                      # Verbose
        "--tb=short",              # Traceback کوتاه
        "--cov=.",                 # Coverage برای پروژه
        "--cov-report=term-missing",  # نمایش خطوط پوشش نشده
        "--cov-report=html",       # گزارش HTML
        "--color=yes"              # رنگی
    ]
    
    try:
        result = subprocess.run(cmd, check=False)
        
        print("\n" + "="*70)
        if result.returncode == 0:
            print("✅ تمام تست‌ها موفق بودند!")
        else:
            print("⚠️  برخی تست‌ها ناموفق بودند")
        print("="*70)
        
        print("\n📊 گزارش Coverage در htmlcov/index.html ذخیره شد")
        print("   برای مشاهده: باز کردن htmlcov/index.html در مرورگر")
        
        return result.returncode == 0
    
    except FileNotFoundError:
        print("\n❌ pytest نصب نیست!")
        print("   نصب: pip install pytest pytest-cov")
        return False


def run_specific_test(test_file: str):
    """اجرای یک تست خاص"""
    
    print(f"\n🧪 اجرای {test_file}...")
    
    cmd = [
        sys.executable,
        "-m",
        "pytest",
        f"tests/{test_file}",
        "-v",
        "--tb=short"
    ]
    
    subprocess.run(cmd)


def run_tests_by_marker(marker: str):
    """اجرای تست‌های با marker خاص"""
    
    print(f"\n🧪 اجرای تست‌های با marker: {marker}...")
    
    cmd = [
        sys.executable,
        "-m",
        "pytest",
        "-v",
        "-m", marker,
        "--tb=short"
    ]
    
    subprocess.run(cmd)


def main():
    """منوی اصلی"""
    
    print("\n" + "="*70)
    print("🧪 سیستم تست ربات معامله‌گر")
    print("="*70)
    print("\n1. اجرای تمام تست‌ها (با Coverage)")
    print("2. تست Config")
    print("3. تست Logger")
    print("4. تست Monitor")
    print("5. تست API")
    print("0. خروج")
    
    choice = input("\nانتخاب کنید (1-5): ").strip()
    
    if choice == "1":
        run_tests()
    elif choice == "2":
        run_specific_test("test_config.py")
    elif choice == "3":
        run_specific_test("test_logger.py")
    elif choice == "4":
        run_specific_test("test_monitor.py")
    elif choice == "5":
        run_specific_test("test_api.py")
    elif choice == "0":
        print("خروج...")
        return
    else:
        print("❌ انتخاب نامعتبر!")
        return
    
    print("\n" + "="*70)
    print("✅ اجرای تست‌ها پایان یافت")
    print("="*70)


if __name__ == "__main__":
    # اگر بدون آرگومان اجرا شد، منو نمایش بده
    if len(sys.argv) == 1:
        main()
    # اگر با --all اجرا شد، همه را اجرا کن
    elif "--all" in sys.argv:
        run_tests()
    # اگر نام فایل داده شد
    elif len(sys.argv) == 2:
        run_specific_test(sys.argv[1])
    else:
        print("استفاده:")
        print("  python run_tests.py           # منوی تعاملی")
        print("  python run_tests.py --all     # اجرای همه")
        print("  python run_tests.py test_config.py  # تست خاص")

            "/api/token/refresh",

        assert response.status_code == 200
        data = response.json()
        assert "access_token" in data

        )

        for i in range(40):
            pred = np.random.rand()
            conf = np.random.rand()  # confidence تصادفی
            direction = 'BUY' if pred > 0.5 else 'SELL'
            
            monitor.log_prediction(pred, conf, direction, f'Test {i}')
            
            # عملکرد بهتر با confidence بالاتر
            accuracy_rate = 0.4 + (conf * 0.4)  # 40% تا 80%
            was_correct = np.random.rand() < accuracy_rate
            
            monitor.log_outcome(i+1, direction, was_correct, 50.0 if was_correct else -50.0)
        
        yield monitor
        shutil.rmtree(temp_dir)

        for label in expected_labels:

                assert 'correct' in analysis[label]
                assert 'accuracy' in analysis[label]
